function drawNebulaBlob(ctx,x,y,radius,color,alpha=.16){
  const g = ctx.createRadialGradient(x,y,0,x,y,radius);
  g.addColorStop(0, color.replace('ALPHA', String(alpha)));
  g.addColorStop(.45, color.replace('ALPHA', String(alpha*.45)));
  g.addColorStop(1, color.replace('ALPHA','0'));
  ctx.fillStyle = g;
  ctx.beginPath();
  ctx.arc(x,y,radius,0,Math.PI*2);
  ctx.fill();
}

function drawDeepSpace(){
  if(!deepSpaceCtx) return;

  const ctx = deepSpaceCtx;
  ctx.clearRect(0,0,WORLD_W,WORLD_H);

  const bg = ctx.createLinearGradient(0,0,WORLD_W,WORLD_H);
  bg.addColorStop(0,'#01030a');
  bg.addColorStop(.35,'#030914');
  bg.addColorStop(1,'#02050d');
  ctx.fillStyle = bg;
  ctx.fillRect(0,0,WORLD_W,WORLD_H);

  const nebulaScale = gameSettings.powerSaver
    ? .7
    : ({low:.8,medium:1,high:1.1,ultra:1.3}[gameSettings.graphics] || 1);

  const blobs = [
    [WORLD_W*.18,WORLD_H*.22,340*nebulaScale,'rgba(78,126,255,ALPHA)',.20],
    [WORLD_W*.76,WORLD_H*.20,300*nebulaScale,'rgba(180,92,255,ALPHA)',.18],
    [WORLD_W*.48,WORLD_H*.64,380*nebulaScale,'rgba(55,204,240,ALPHA)',.15],
    [WORLD_W*.84,WORLD_H*.74,260*nebulaScale,'rgba(255,126,86,ALPHA)',.12],
    [WORLD_W*.26,WORLD_H*.78,220*nebulaScale,'rgba(89,255,192,ALPHA)',.08]
  ];
  blobs.forEach(b => drawNebulaBlob(ctx,...b));

  const dustCount = gameSettings.powerSaver
    ? 70
    : ({low:90,medium:150,high:220,ultra:280}[gameSettings.graphics] || 150);

  for(let i=0;i<dustCount;i++){
    const x = Math.random()*WORLD_W;
    const y = Math.random()*WORLD_H;
    const r = Math.random()*1.9 + .25;

    ctx.fillStyle = Math.random() < .6
      ? 'rgba(255,255,255,.12)'
      : (Math.random()<.5
          ? 'rgba(169,205,255,.10)'
          : 'rgba(255,222,188,.10)');

    ctx.beginPath();
    ctx.arc(x,y,r,0,Math.PI*2);
    ctx.fill();
  }

  const streaks = gameSettings.powerSaver
    ? 12
    : ({low:14,medium:20,high:28,ultra:38}[gameSettings.graphics] || 20);

  ctx.lineCap = 'round';

  for(let i=0;i<streaks;i++){
    const x = Math.random()*WORLD_W;
    const y = Math.random()*WORLD_H;
    const len = Math.random()*40 + 12;
    const ang = (Math.random()*Math.PI/2) - Math.PI/4;

    ctx.strokeStyle = 'rgba(255,255,255,.06)';
    ctx.lineWidth = Math.random()*1.2 + .3;
    ctx.beginPath();
    ctx.moveTo(x,y);
    ctx.lineTo(x + Math.cos(ang)*len, y + Math.sin(ang)*len);
    ctx.stroke();
  }
}

function generateBalancedPlanets(){
  const planets = [];
  const starts = makeStartPlanets();
  planets.push(...starts);

  const allStyles = Object.keys(STYLE_DEFS);
  const habitableStyles = [...HABITABLE_STYLES].filter(key => allStyles.includes(key));

  function addNearbyColonizablePlanet(startPlanet, side){
    // Гарантированная стартовая колония: близко, но не вплотную к базе.
    // Координаты каждый раз немного отличаются, поэтому стартовые зоны не зеркальны.
    for(let attempt=0; attempt<240; attempt++){
      const styleKey = habitableStyles[randi(0, habitableStyles.length-1)];
      const inwardAngle = side === 'player' ? -Math.PI/2 : Math.PI/2;
      const angle = inwardAngle + rand(-0.72, 0.72);
      const distance = rand(350, 470);
      const px = startPlanet.x + Math.cos(angle) * distance;
      const py = startPlanet.y + Math.sin(angle) * distance;

      if(px < 180 || px > WORLD_W-180 || py < 220 || py > WORLD_H-220) continue;

      const candidate = createPlanet(
        px,
        py,
        rand(44, 54),
        styleKey,
        'neutral',
        STYLE_DEFS[styleKey].title + ' — ближняя колония',
        false
      );
      candidate.habitable = true;

      let ok = true;
      for(const existing of planets){
        if(circlesOverlap(candidate, existing, 105)){ ok = false; break; }
      }
      if(ok){
        planets.push(candidate);
        return candidate;
      }
    }
    return null;
  }

  // Симметрия только по возможностям, а не по координатам:
  // у каждой стороны гарантированно одна пригодная для колонизации планета рядом со стартом.
  addNearbyColonizablePlanet(starts[1], 'player');
  addNearbyColonizablePlanet(starts[0], 'enemy');

  let attempts = 0;
  const targetTotalBeforeSpecials = 18;
  while(planets.length < targetTotalBeforeSpecials && attempts < 12000){
    attempts++;
    const styleKey = allStyles[randi(0, allStyles.length-1)];
    const x = rand(260, WORLD_W-260);
    const y = rand(380, WORLD_H-380);
    const r = rand(40, 72);
    const candidate = createPlanet(x, y, r, styleKey, 'neutral', STYLE_DEFS[styleKey].title + ' ' + (planets.length-1));
    let ok = true;
    for(const existing of planets){
      if(circlesOverlap(candidate, existing, 105)){ ok = false; break; }
    }
    if(!ok) continue;
    const distPlayer = Math.hypot(candidate.x - starts[1].x, candidate.y - starts[1].y);
    const distEnemy = Math.hypot(candidate.x - starts[0].x, candidate.y - starts[0].y);
    // Стартовая зона зарезервирована под гарантированные объекты каждой стороны.
    // Остальные планеты начинают появляться уже за её пределами.
    if(distPlayer < 650 || distEnemy < 650) continue;
    planets.push(candidate);
  }

  const specials = [
    createPlanet(rand(620, 1040), rand(940, 1320), 62, 'cream_ringed', 'neutral', 'Кольцевая цитадель'),
    createPlanet(rand(2050, 2580), rand(820, 1260), 58, 'galaxy_purple', 'neutral', 'Туманная сфера'),
    createPlanet(rand(1220, 1700), rand(470, 860), 52, 'moon_gray', 'neutral', 'Лунная шахта'),
    createPlanet(rand(1480, 2140), rand(1620, 2160), 52, 'iridescent', 'neutral', 'Переливчатое ядро')
  ];
  for(const sp of specials){
    let safe = true;
    for(const existing of planets){
      if(circlesOverlap(sp, existing, 130)){ safe = false; break; }
    }
    if(safe) planets.push(sp);
  }
  return planets;
}

function generateResearchAsteroids(){

      const a = {
        id:'asteroid-' + researchAsteroids.length,
        kind:'researchAsteroid',
        x:startPlanet.x + Math.cos(angle) * distance,
        y:startPlanet.y + Math.sin(angle) * distance,
        r:22,
        name:'Астероид ' + (researchAsteroids.length+1),
        playerScanned:true,
        reserve:randi(300,520),
        starterZone:side
      };

      if(a.x < 180 || a.x > WORLD_W-180 || a.y < 220 || a.y > WORLD_H-220) continue;

      const farFromAsteroids = researchAsteroids.every(b => Math.hypot(a.x-b.x,a.y-b.y) > 230);
      if(farFromPlanets && farFromAsteroids){
        researchAsteroids.push(a);
        return a;
      }
    }
    return null;
  }

  // Равный старт: по одному астероиду рядом с игроком и противником.
  addNearbyStarterAsteroid(playerPlanet, 'player');
  addNearbyStarterAsteroid(enemyPlanet, 'enemy');

  // Остальные астероиды по-прежнему распределяются хаотично по всей карте.

  while(researchAsteroids.length < 8 && tries < 5000){
    tries++;

      id:'asteroid-' + researchAsteroids.length,
      kind:'researchAsteroid',
      x:rand(250,WORLD_W-250),
      y:rand(380,WORLD_H-380),
      r:22,

    };

  }

function worldViewportBounds(margin=220){
  const inv = 1 / Math.max(.01,scale);
  return {
    left:(-x)*inv - margin,
    top:(-y)*inv - margin,
    right:(innerWidth-x)*inv + margin,
    bottom:(innerHeight-y)*inv + margin
  };
}
function isWorldPointNearViewport(px,py,margin=180){
  const b = worldViewportBounds(margin);
  return px >= b.left && px <= b.right && py >= b.top && py <= b.bottom;
}
function updateWorldCulling(force=false){
  const now = performance.now();
  if(!force && !PERF.cullingDirty && now-PERF.cullLast < PERF.cullInterval) return;
  if(!force && now-PERF.cullLast < PERF.cullInterval) return;
  PERF.cullLast = now;
  PERF.cullingDirty = false;

  const b = worldViewportBounds(300);
  const inside = (px,py,r=0) =>
    px+r >= b.left && px-r <= b.right && py+r >= b.top && py-r <= b.bottom;

  planets.forEach(p=>{
    const visible = inside(p.x,p.y,Math.max(90,p.r*3));
    if(p.wrapEl) p.wrapEl.style.display = visible ? '' : 'none';
    if(p.auraEl) p.auraEl.style.display = visible ? '' : 'none';
    if(p.hitEl) p.hitEl.style.display = visible ? '' : 'none';
    if(p.labelEl) p.labelEl.style.display = visible ? '' : 'none';
    if(p.tagEl) p.tagEl.style.display = visible ? '' : 'none';
  });

  researchAsteroids.forEach(a=>{
    if(a.el) a.el.style.display = inside(a.x,a.y,110) ? '' : 'none';
  });
}

function isEnemyShipVisibleOnMini(ship){
  if(ship.owner !== 'enemy') return false;
  if(!gameSettings.fog) return true;
  if(ship.combatTarget && ship.combatTarget.owner === 'player') return true;
  if(ship.targetPlanet && ship.targetPlanet.owner === 'player') return true;

}

function updateMiniDynamic(force=false){
  if(gameSettings.minimap === 'off') return;

  const now = performance.now();
  if(!force && now - PERF.miniLast < PERF.miniInterval) return;
  PERF.miniLast = now;

  const mw = minimap.clientWidth, mh = minimap.clientHeight;
  const dots = [];
  const alerts = [];

  for(const ship of scouts){
    if(ship.destroyed || !isEnemyShipVisibleOnMini(ship)) continue;
    dots.push(
      '<span class="miniShipDot enemy" style="left:' +
      ((ship.x/WORLD_W)*mw).toFixed(1) + 'px;top:' +
      ((ship.y/WORLD_H)*mh).toFixed(1) + 'px"></span>'
    );
  }

  for(const planet of planets){
    if(planet.owner !== 'player' || !planet.captureState) continue;

      '<span class="miniAlertPing" style="left:' +
      ((planet.x/WORLD_W)*mw).toFixed(1) + 'px;top:' +
      ((planet.y/WORLD_H)*mh).toFixed(1) + 'px"></span>'
    );
  }

  for(const ship of scouts){
    if(ship.owner !== 'player') continue;
    if(!ship.combatTarget && !ship.attackAlerted) continue;
    alerts.push(
      '<span class="miniAlertPing" style="left:' +
      ((ship.x/WORLD_W)*mw).toFixed(1) + 'px;top:' +
      ((ship.y/WORLD_H)*mh).toFixed(1) + 'px"></span>'

  }

  const alertsHtml = alerts.join('');

  if(updateMiniDynamic._dotsHtml !== dotsHtml){
    miniDynamicDots.innerHTML = dotsHtml;
    updateMiniDynamic._dotsHtml = dotsHtml;
  }
  if(updateMiniDynamic._alertsHtml !== alertsHtml){

  }

const SCOUT_SENSOR_RANGE = 720; // увеличенная дальность спутника-разведчика

function isExploredPoint(px,py){
  return exploredAreas.some(a => Math.hypot(px-a.x, py-a.y) <= a.r * .92);
}

function isPlanetDiscovered(planet){
  return exploredAreas.some(a => {
    const d = Math.hypot(planet.x-a.x, planet.y-a.y);
    return d <= a.r + Math.max(22, planet.r * .75);
  });
}

function canScoutTargetPlanet(scout, planet){
  if(planet.kind === 'researchAsteroid'){
    return !!scout.canMineResearch && isExploredPoint(planet.x,planet.y);
  }

  if(isPlanetDiscovered(planet)) return true;

  if(!scout.canScan) return false;

  // Только разведчики могут обнаруживать соседние планеты через туман.
  const origin = scout.orbitPlanet || scout;
  return Math.hypot(planet.x-origin.x, planet.y-origin.y) <= SCOUT_SENSOR_RANGE;
}

function revealArea(px,py,radius){
  exploredAreas.push({x:px,y:py,r:radius});
  drawFog();
}

function paintMainFogHole(a){
  const s = FOG_RENDER_SCALE;
  const ax = a.x*s, ay = a.y*s, ar = a.r*s;
  const grad = fogCtx.createRadialGradient(ax,ay,ar*.58,ax,ay,ar);
  grad.addColorStop(0,'rgba(0,0,0,1)');
  grad.addColorStop(.78,'rgba(0,0,0,.96)');
  grad.addColorStop(1,'rgba(0,0,0,0)');
  fogCtx.fillStyle = grad;
  fogCtx.beginPath();
  fogCtx.arc(ax,ay,ar,0,Math.PI*2);
  fogCtx.fill();
}

function paintMiniFogHole(a,mw,mh){
  const mx = (a.x/WORLD_W)*mw;
  const my = (a.y/WORLD_H)*mh;
  const mr = (a.r/WORLD_W)*mw;
  const grad = miniFogCtx.createRadialGradient(mx,my,mr*.58,mx,my,mr);
  grad.addColorStop(0,'rgba(0,0,0,1)');
  grad.addColorStop(.78,'rgba(0,0,0,.96)');
  grad.addColorStop(1,'rgba(0,0,0,0)');
  miniFogCtx.fillStyle = grad;
  miniFogCtx.beginPath();
  miniFogCtx.arc(mx,my,mr,0,Math.PI*2);
  miniFogCtx.fill();
}

function drawFog(forceFull=false){
  const mw = Math.max(1,minimap.clientWidth);
  const mh = Math.max(1,minimap.clientHeight);

  if(!gameSettings.fog){
    if(PERF.fogMode !== 'off' || forceFull){
      fogCtx.setTransform(1,0,0,1,0,0);
      fogCtx.globalCompositeOperation = 'source-over';
      fogCtx.clearRect(0,0,fogCanvas.width,fogCanvas.height);

      if(miniFog.width !== mw) miniFog.width = mw;
      if(miniFog.height !== mh) miniFog.height = mh;
      miniFogCtx.clearRect(0,0,mw,mh);

      PERF.fogRenderedCount = 0;
      PERF.miniFogRenderedCount = 0;
      PERF.miniFogW = mw;
      PERF.miniFogH = mh;
      PERF.fogMode = 'off';
    }
    updateMiniDynamic(true);
    return;
  }

  const mainFull =
    forceFull ||
    PERF.fogMode !== 'on' ||
    PERF.fogRenderedCount > exploredAreas.length;

  if(mainFull){
    fogCtx.globalCompositeOperation = 'source-over';
    fogCtx.clearRect(0,0,fogCanvas.width,fogCanvas.height);
    fogCtx.fillStyle = 'rgb(2,7,16)';
    fogCtx.fillRect(0,0,fogCanvas.width,fogCanvas.height);
    PERF.fogRenderedCount = 0;
  }

  if(PERF.fogRenderedCount < exploredAreas.length){
    fogCtx.globalCompositeOperation = 'destination-out';
    for(let i=PERF.fogRenderedCount;i<exploredAreas.length;i++){
      paintMainFogHole(exploredAreas[i]);
    }
    fogCtx.globalCompositeOperation = 'source-over';
    PERF.fogRenderedCount = exploredAreas.length;
  }

  const miniSizeChanged = PERF.miniFogW !== mw || PERF.miniFogH !== mh;
  if(miniFog.width !== mw) miniFog.width = mw;
  if(miniFog.height !== mh) miniFog.height = mh;

  const miniFull =
    forceFull ||
    PERF.fogMode !== 'on' ||
    miniSizeChanged ||
    PERF.miniFogRenderedCount > exploredAreas.length;

  if(miniFull){
    miniFogCtx.globalCompositeOperation = 'source-over';
    miniFogCtx.clearRect(0,0,mw,mh);
    miniFogCtx.fillStyle = 'rgb(2,7,16)';
    miniFogCtx.fillRect(0,0,mw,mh);
    PERF.miniFogRenderedCount = 0;
    PERF.miniFogW = mw;
    PERF.miniFogH = mh;
  }

  if(PERF.miniFogRenderedCount < exploredAreas.length){
    miniFogCtx.globalCompositeOperation = 'destination-out';
    for(let i=PERF.miniFogRenderedCount;i<exploredAreas.length;i++){
      paintMiniFogHole(exploredAreas[i],mw,mh);
    }
    miniFogCtx.globalCompositeOperation = 'source-over';
    PERF.miniFogRenderedCount = exploredAreas.length;
  }

  PERF.fogMode = 'on';
}

let pinchPointers = new Map(), pinchStartDistance = 0, pinchStartScale = 1, pinchCenterX = 0, pinchCenterY = 0;
function uiHit(target){ return minimap.contains(target) || target.closest('.planetHit') || target.closest('.scout') || target.closest('#planetFleetOverlay') || target.closest('#shipControlOverlay'); }

viewport.addEventListener('pointerdown', e => {
  if(uiHit(e.target)) return;
  if(e.pointerType==='touch'){
    pinchPointers.set(e.pointerId, {x:e.clientX,y:e.clientY});
    if(pinchPointers.size===2){
      const pts=[...pinchPointers.values()];
      pinchStartDistance=Math.hypot(pts[1].x-pts[0].x, pts[1].y-pts[0].y);
      pinchStartScale=scale; pinchCenterX=(pts[0].x+pts[1].x)/2; pinchCenterY=(pts[0].y+pts[1].y)/2;
      pan=false; panPointer=null; return;
    }
  }
  if(e.pointerType==='mouse' && e.button!==0) return;
  e.preventDefault(); pan=true; panMoved=false; panPointer=e.pointerId; panStartX=e.clientX; panStartY=e.clientY; worldStartX=x; worldStartY=y;
  try{ viewport.setPointerCapture(e.pointerId); }catch(err){}
}, {passive:false});
viewport.addEventListener('pointermove', e => {
  if(e.pointerType==='touch' && pinchPointers.has(e.pointerId)){
    pinchPointers.set(e.pointerId, {x:e.clientX,y:e.clientY});
    if(pinchPointers.size===2){
      e.preventDefault(); const pts=[...pinchPointers.values()]; const d=Math.hypot(pts[1].x-pts[0].x, pts[1].y-pts[0].y);
      if(pinchStartDistance>0) setZoom(pinchStartScale*(d/pinchStartDistance), pinchCenterX, pinchCenterY);
      return;
    }
  }
  if(!pan || e.pointerId!==panPointer) return;
  if(Math.hypot(e.clientX-panStartX,e.clientY-panStartY) > 8) panMoved=true;
  e.preventDefault(); x=worldStartX+(e.clientX-panStartX)*cameraPanFactor; y=worldStartY+(e.clientY-panStartY)*cameraPanFactor; render();
}, {passive:false});
function clearPinchPointer(e){ if(e && pinchPointers.has(e.pointerId)) pinchPointers.delete(e.pointerId); if(pinchPointers.size < 2) pinchStartDistance = 0; }
function endPan(e){
  clearPinchPointer(e);
  if(e && panPointer !== null && e.pointerId !== panPointer) return;
  const wasTap = !!e && pan && !panMoved && pinchPointers.size === 0;
  pan = false; panPointer = null;
  if(!wasTap) return;

  const p = screenToWorldPoint(e.clientX,e.clientY);
  if(pointCommandShips.length){
    sendShipsToPoint(pointCommandShips,p.x,p.y);
    pointCommandShips = [];
    clearSelectedFleet();
    clearScoutSelection();
    return;
  }
  if(selectedFleet.length && selectedFleet.every(s => s.type !== 'scout')){
    sendShipsToPoint([...selectedFleet],p.x,p.y);
    clearSelectedFleet();
    clearScoutSelection();
  }

function gameTimeText(){
  const total = Math.max(0, Math.floor(gameElapsed));
  const m = Math.floor(total / 60);
  const s = total % 60;
  return String(m).padStart(2,'0') + ':' + String(s).padStart(2,'0');
}


function ensureEventAudio(){
  if(!gameSettings.eventSounds) return null;
  try{
    if(!eventAudioContext){
      const Ctx = window.AudioContext || window.webkitAudioContext;
      if(Ctx) eventAudioContext = new Ctx();
    }
    if(eventAudioContext?.state === 'suspended') eventAudioContext.resume().catch(()=>{});
    return eventAudioContext;
  }catch(err){ return null; }
}
function playEventSound(kind){
  if(!gameSettings.eventSounds || gameSettings.eventVolume <= 0) return;
  const ctx = ensureEventAudio();
  if(!ctx) return;
  try{
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    const now = ctx.currentTime;
    const volume = Math.max(.001,(gameSettings.eventVolume/100)*.055);
    const freq = kind === 'attack' ? 240 : kind === 'detection' ? 420 : kind === 'success' ? 690 : 520;
    osc.type = kind === 'attack' ? 'sawtooth' : 'sine';
    osc.frequency.setValueAtTime(freq,now);
    if(kind === 'attack') osc.frequency.exponentialRampToValueAtTime(155,now+.18);
    gain.gain.setValueAtTime(volume,now);
    gain.gain.exponentialRampToValueAtTime(.001,now+.22);
    osc.connect(gain); gain.connect(ctx.destination);
    osc.start(now); osc.stop(now+.24);
  }catch(err){}
}
function vibrateForEvent(kind){
  if(!gameSettings.vibration || !navigator.vibrate) return;
  if(kind === 'attack') navigator.vibrate([70,45,100]);
  else if(kind === 'detection') navigator.vibrate([45,40,45]);
  else if(kind === 'success') navigator.vibrate(35);
}
function triggerAutoSlow(kind){
  if(!gameSettings.autoSlow || !gameStarted || !['attack','detection'].includes(kind)) return;
  if(autoSlowTimer) clearTimeout(autoSlowTimer);
  if(autoSlowPreviousSpeed === null) autoSlowPreviousSpeed = gameSpeed;
  setGameSpeed(.5,false);
  autoSlowTimer = setTimeout(()=>{
    if(autoSlowPreviousSpeed !== null){
      setGameSpeed(autoSlowPreviousSpeed,false);
      autoSlowPreviousSpeed = null;
    }
    autoSlowTimer = null;
  },8000);
}

function pushEvent(message, kind='normal'){
  const item = document.createElement('div');
  item.className = 'eventItem ' + kind;

  const time = document.createElement('span');
  time.className = 'eventTime';
  time.textContent = gameTimeText();

  const body = document.createElement('span');
  body.textContent = message;

  item.appendChild(time);
  item.appendChild(body);
  eventLogItems.appendChild(item);

  while(eventLogItems.children.length > 80){
    eventLogItems.removeChild(eventLogItems.firstChild);
  }
  eventLogItems.scrollTop = eventLogItems.scrollHeight;

  if(!eventLogPanel.classList.contains('open')){
    unreadEvents++;
    eventLogBadge.textContent = unreadEvents > 99 ? '99+' : String(unreadEvents);
    eventLogBadge.style.display = 'block';
  }

  if(kind === 'detection') setAlertLevel('orange');
  if(kind === 'attack') setAlertLevel('red');

  if(['attack','detection','success'].includes(kind)){
    playEventSound(kind);
    vibrateForEvent(kind);
    triggerAutoSlow(kind);
  }
}

function releaseShipSignalAssignment(ship){
  if(!ship || !ship._aeAnomalyTargetId) return null;
  const ae=window.AELivingGalaxy;
  const a=ae?.anomalies?.find?.(x=>x.id===ship._aeAnomalyTargetId) || null;
  if(a){
    if(a.targetedBy===ship.id) a.targetedBy=null;
    if(a.readyShipId===ship.id) a.readyShipId=null;
    a._autoLootOpened=false;
    a._autoLootPending=false;
    a.el?.classList.remove('targeted','ready');
    if(!a.resolved && a.el){
      a.el.innerHTML='?<span class="aeAnomalyLabel">НЕИЗВЕСТНЫЙ СИГНАЛ</span>';
    }
  }
  ship._aeAnomalyTargetId=null;
  ship._aeAnomalyWaiting=false;
  return a;
}
function nearestOwnedColonyToShip(ship){
  let best = null, bestD = Infinity;
  for(const p of planets){
    if(p.owner !== 'player') continue;
    const d = Math.hypot(p.x-ship.x,p.y-ship.y);
    if(d < bestD){ bestD = d; best = p; }
  }
  return best || playerPlanet;
}

function shipStatusText(ship){
  if(ship.cooldownState){
    const total = Math.max(1, ship.cooldownState.total || 120);
    const percent = Math.max(0, Math.min(100, Math.round((1 - ship.cooldownState.remaining / total) * 100)));
    return 'Заряд солнечных батарей • ' + percent + '%';
  }
  if(ship.scanState){ const n = activePlanetScanners(ship.scanState.planet, ship.owner).length; return 'Исследование планеты • ' + Math.ceil(ship.scanState.remaining) + ' сек.' + (n > 1 ? ' • ×' + n : ''); }
  if(ship.resourceTransport){
    const m = ship.resourceTransport;
    const target = m.stage === 'returning' ? m.sourcePlanet : m.targetPlanet;
    return (m.stage === 'returning' ? 'Возвращает груз' : 'Доставка ресурсов') +
      ' • ' + m.total + ' ед. → «' + (target?.name || 'цель') + '»';
  }
  if(ship.planetMining){
    const m = ship.planetMining;
    if(m.stage === 'mining') return 'Загружает ресурсы на «' + m.planet.name + '» • ' + Math.ceil(m.remaining) + ' сек.';
    if(m.stage === 'returning') return 'Возвращается с ресурсами на «' + m.homePlanet.name + '»';
    return 'Грузовой рейс к «' + m.planet.name + '»';
  }
  if(ship.combatTarget) return 'В бою с «' + ship.combatTarget.name + '»';
  if(ship._aeAnomalyWaiting) return '✦ У неизвестного сигнала • находка готова к осмотру';
  if(ship._aeAnomalyTargetId) return 'Летит к неизвестному сигналу';
  if(ship.targetPoint) return 'Летит к указанной точке';
  if(ship.patrolState){
    if(ship.patrolState.returning) return ship.autoPatrolPlanet
      ? 'Возвращается в патруль вокруг «' + ship.autoPatrolPlanet.name + '»'
      : 'Возвращается в зону патруля';
    return ship.autoPatrolPlanet
      ? 'Автопатруль вокруг «' + ship.autoPatrolPlanet.name + '»'
      : 'Патрулирует область';
  }
  if(ship.targetPlanet){
    const targetName = ship.targetPlanet.name || 'цель';
    return 'В полёте к «' + targetName + '»';
  }
  if(ship.stationary) return 'Остановлен в открытом космосе';
  if(ship.researchMining) return 'Добывает Ксенит';
  if(ship.orbitPlanet) return 'На орбите «' + (ship.orbitPlanet.name || 'объекта') + '»';
  return 'Ожидает приказа';
}

function returnShip(ship){
  if(!ship || ship.scanState || ship.cooldownState) return;

  if(ship._aeAnomalyTargetId){
    const wasWaiting=!!ship._aeAnomalyWaiting;
    releaseShipSignalAssignment(ship);
    const home=(ship.homePlanet?.owner==='player' ? ship.homePlanet : nearestOwnedColonyToShip(ship));
    if(home){
      ship.stationary=false;
      setScoutTarget(ship,home,{silent:true});
      pushEvent(ship.name+(wasWaiting?' оставил находку у сигнала':' отменил экспедицию')+' и возвращается на «'+home.name+'».','discovery');
      showHint('«'+ship.name+'» возвращается на «'+home.name+'». Неизвестный сигнал снова доступен.');
      return;
    }
  }

  if(ship.resourceTransport){
    const mission = ship.resourceTransport;
    mission.stage = 'returning';
    ship.stationary = false;
    ship.patrolState = null;
    ship.combatTarget = null;
    removePatrolVisual(ship);
    ship.targetPoint = null;
    clearWaypointMarker(ship);

    if(ship.orbitPlanet === mission.sourcePlanet){
      finishResourceTransportArrival(ship,mission.sourcePlanet);
      return;
    }

    setScoutTarget(ship,mission.sourcePlanet,{silent:true,preserveTransport:true});
    pushEvent(
      ship.name + ': доставка отменена, корабль возвращает груз на «' + mission.sourcePlanet.name + '».',
      'discovery'
    );
    return;
  }

  const miningMissionActive = !!(ship.researchMining || ship.planetMining);
  const returnTarget = (miningMissionActive && ship.homePlanet) ? ship.homePlanet : ship.lastDepartureTarget;
  if(!returnTarget) return;

  // Для добывающего рейса команда «Возврат» означает полный отзыв:
  // корабль возвращается на базу, выгружает груз и больше сам не улетает.
  const recallingAsteroidMiner = !!(ship.canMineResearch && ship.researchMining);
  const recallingPlanetMiner = !!ship.planetMining;
  if(recallingAsteroidMiner){
    ship.researchMining.loopActive = false;
    ship.researchMining.stage = 'returning';
  }
  if(recallingPlanetMiner){
    ship.planetMining.loopActive = false;
    ship.planetMining.stage = 'returning';
  }

  ship.stationary = false;
  ship.autoPatrolPlanet = null;
  ship.patrolState = null;
  ship.combatTarget = null;
  removePatrolVisual(ship);
  ship.targetPoint = null; clearWaypointMarker(ship);

  setScoutTarget(
    ship,
    returnTarget,
    recallingAsteroidMiner
      ? {preserveMining:true}
      : recallingPlanetMiner
        ? {preservePlanetMining:true}
        : {}
  );

  if(recallingAsteroidMiner){
    pushEvent(ship.name + ' отозван с астероида и возвращается на базу.', 'discovery');
  }else if(recallingPlanetMiner){
    pushEvent(ship.name + ' отозван с добывающей экспедиции и возвращается на базу.', 'discovery');
  }
}

function enablePatrol(ship,options={}){
  if(!ship || ship.scanState || ship.cooldownState) return false;
  if(ship.resourceTransport){
    if(!options.silent){
      pushEvent('Сначала завершите доставку ресурсов или верните грузовой корабль на базу.', 'discovery');
    }
    return false;
  }

  if(ship.owner === 'enemy' && ship.targetPlanet?.id){
    enemyReservedPlanetIds.delete(ship.targetPlanet.id);
  }

  const orbitBeforePatrol = ship.orbitPlanet;
  const patrolCenter = options.aroundPlanet || ((orbitBeforePatrol && orbitBeforePatrol.owner===ship.owner) ? orbitBeforePatrol : null);

  releaseShipSignalAssignment(ship);
  ship.targetPlanet = null;
  ship.targetPoint = null; clearWaypointMarker(ship);
  ship.orbitPlanet = null;
  ship.researchMining = null;
  ship.planetMining = null;
  ship.planetMiningCargo = {};
  ship.combatTarget = null;
  ship.stationary = false;
  ship.autoPatrolPlanet = patrolCenter || null;
  if(patrolCenter && !ship.homePlanet) ship.homePlanet=patrolCenter;
  const patrolRadius = options.radius || 175;
  ship.patrolState = {
    cx:patrolCenter ? patrolCenter.x : ship.x,
    cy:patrolCenter ? patrolCenter.y : ship.y,
    radius:patrolRadius,
    chaseRadius:options.chaseRadius || Math.max(patrolRadius + 110, patrolRadius * 1.65),
    orbitRadius:54,
    angle:Math.random()*Math.PI*2,
    direction:Math.random()<.5 ? -1 : 1,
    detectionCooldown:0,
    returning:false,
    chasing:false,
    chaseTargetId:null
  };
  ensurePatrolVisual(ship);

  if(!options.silent){
    pushEvent(ship.name + ' переведён в режим патруля.', 'discovery');
  }
  return true;
}

function orbitRadiusFor(planet){
  return Math.max(planet.r + 18, 42);
}

function placeScoutOnOrbit(scout){
  if(!scout.orbitPlanet) return;
  const planet = scout.orbitPlanet;
  const radius = orbitRadiusFor(planet);
  scout.x = planet.x + Math.cos(scout.orbitAngle) * radius;
  scout.y = planet.y + Math.sin(scout.orbitAngle) * radius * 0.62;
}

function updateScoutElement(scout){
  if(!scout?.el) return;

  const onScreen = isWorldPointNearViewport(scout.x,scout.y,190);
  if(!onScreen){
    if(!scout._culled){
      scout.el.style.visibility = 'hidden';
      if(scout.timerEl) scout.timerEl.style.visibility = 'hidden';
      scout._culled = true;
    }
    return;
  }

  if(scout._culled){
    scout.el.style.visibility = 'visible';
    if(scout.timerEl) scout.timerEl.style.visibility = 'visible';
    scout._culled = false;
  }

  // Вся корабельная графика (кроме спутника-разведчика) нарисована носом вверх.
  // Математический угол движения 0° смотрит вправо, поэтому для изображений кораблей
  // нужен постоянный визуальный поворот +90°. Направление берём прежде всего из
  // реального вектора движения — так корабль корректно смотрит вперёд при перелёте,
  // патруле, погоне, возвращении с добычи и движении по орбите.
  let travelAngle = Number.isFinite(scout._headingDeg) ? scout._headingDeg : null;

  if(travelAngle === null && scout.combatTarget && !scout.combatTarget.destroyed){
    travelAngle = Math.atan2(
      scout.combatTarget.y - scout.y,
      scout.combatTarget.x - scout.x
    ) * 180 / Math.PI;
  }else if(travelAngle === null && scout.targetPoint){
    travelAngle = Math.atan2(
      scout.targetPoint.y - scout.y,
      scout.targetPoint.x - scout.x
    ) * 180 / Math.PI;
  }else if(travelAngle === null && scout.targetPlanet){
    travelAngle = Math.atan2(
      scout.targetPlanet.y - scout.y,
      scout.targetPlanet.x - scout.x
    ) * 180 / Math.PI;
  }else if(travelAngle === null && scout.orbitPlanet){
    travelAngle = (scout.orbitAngle * 180 / Math.PI) +
      (scout.orbitDirection > 0 ? 90 : -90);
  }

  let angle = (travelAngle ?? 0) + (scout.type === 'scout' ? 0 : 90);

  if(scout.orbitPlanet){
    const nextZ = Math.sin(scout.orbitAngle) > 0 ? '5' : '1';
    if(scout._lastZ !== nextZ){
      scout.el.style.zIndex = nextZ;
      scout._lastZ = nextZ;
    }
  }

  if(!scout._gpuInit){
    scout.el.style.left = '0px';
    scout.el.style.top = '0px';
    scout.el.style.willChange = 'transform';
    if(scout.timerEl){
      scout.timerEl.style.left = '0px';
      scout.timerEl.style.top = '0px';
      scout.timerEl.style.willChange = 'transform';
    }
    scout._gpuInit = true;
  }

  const transform =
    'translate3d(' + scout.x.toFixed(2) + 'px,' + scout.y.toFixed(2) +
    'px,0) translate(-50%,-50%) rotate(' + angle.toFixed(1) + 'deg)';

  if(scout._lastTransform !== transform){
    scout.el.style.transform = transform;
    scout._lastTransform = transform;
  }

  if(!scout.timerEl) return;

  if(scout.scanState){
    scout.timerEl.classList.remove('battery');
    scout.timerEl.style.display = 'block';
    scout.timerEl.style.transform =
      'translate3d(' + scout.x.toFixed(2) + 'px,' + (scout.y-16).toFixed(2) +
      'px,0) translate(-50%,-100%)';

    const value = Math.max(0,Math.ceil(scout.scanState.remaining)) + 'с';
    if(scout._timerContent !== value){
      scout.timerEl.textContent = value;
      scout._timerContent = value;
    }
  }else if(scout.cooldownState){
    scout.timerEl.classList.add('battery');
    scout.timerEl.style.display = 'block';
    scout.timerEl.style.transform =
      'translate3d(' + scout.x.toFixed(2) + 'px,' + (scout.y-16).toFixed(2) +
      'px,0) translate(-50%,-100%)';

    const total = Math.max(1,scout.cooldownState.total || 120);
    const remaining = Math.max(0,scout.cooldownState.remaining);
    const chargePercent = Math.max(0,Math.min(100,Math.round((1-remaining/total)*100)));

    if(scout._batteryPercent !== chargePercent){
      const chargeClass = chargePercent < 20 ? 'low' : (chargePercent < 50 ? 'mid' : '');
      scout.timerEl.innerHTML =
        '<div class="batteryChargeIndicator">' +
          '<span class="batteryShell">' +
            '<span class="batteryFill ' + chargeClass + '" style="width:' + chargePercent + '%"></span>' +
          '</span>' +
          '<span class="batteryPercent">' + chargePercent + '%</span>' +
        '</div>';
      scout._batteryPercent = chargePercent;
      scout._timerContent = null;
    }
  }else{
    if(scout.timerEl.style.display !== 'none') scout.timerEl.style.display = 'none';
    scout._timerContent = null;
    scout._batteryPercent = null;
  }
}

function beginOrbit(scout, planet){
  scout.targetPlanet = null;
  scout.orbitPlanet = planet;
  scout.orbitAngle = Math.atan2(
    (scout.y - planet.y) / 0.62,
    scout.x - planet.x
  );

  if(!Number.isFinite(scout.orbitAngle)){
    scout.orbitAngle = Math.random() * Math.PI * 2;
  }

  placeScoutOnOrbit(scout);
  updateScoutElement(scout);
}



function activePlanetScanners(planet, owner){
  return scouts.filter(s =>
    !s.destroyed &&
    s.owner === owner &&
    s.scanState &&
    s.scanState.planet === planet
  );
}

function syncPlanetScanState(scanners){
  if(!scanners.length) return;
  const remaining = Math.min(...scanners.map(s => s.scanState.remaining));
  const revealed = Math.max(...scanners.map(s => s.scanState.revealed || 0));
  scanners.forEach(s => {
    s.scanState.remaining = remaining;
    s.scanState.revealed = revealed;
  });
}

function startPlanetScan(scout, planet){
  if(!scout.canScan || planet.kind === 'researchAsteroid'){
    scout.scanState = null;
    return;
  }

  if(scout.owner === 'player'){
    if(planet.playerScanned){
      scout.scanState = null;
      return;
    }

    const alreadyScanning = activePlanetScanners(planet, 'player').filter(s => s !== scout);
    const sharedRemaining = alreadyScanning.length
      ? Math.min(...alreadyScanning.map(s => s.scanState.remaining))
      : 60;
    const sharedRevealed = alreadyScanning.length
      ? Math.max(...alreadyScanning.map(s => s.scanState.revealed || 0))
      : 0;

    scout.scanState = {
      planet,
      total:60,
      remaining:sharedRemaining,
      revealed:sharedRevealed
    };

    const count = alreadyScanning.length + 1;
    if(count === 1){
      pushEvent(
        scout.name + ': начато изучение неизвестной планеты. Полное сканирование займёт 60 сек.',
        'discovery'
      );
    }else{
      pushEvent(
        scout.name + ' присоединился к исследованию. Разведчиков у планеты: ' + count +
        '. Скорость сканирования ×' + count + '.',
        'discovery'
      );
    }
  }else{
    if(enemyVisitedPlanetIds.has(planet.id)){
      scout.scanState = null;
      scout.idleTimer = 8;
      return;
    }

    const alreadyScanning = activePlanetScanners(planet, 'enemy').filter(s => s !== scout);
    const sharedRemaining = alreadyScanning.length
      ? Math.min(...alreadyScanning.map(s => s.scanState.remaining))
      : 60;
    const sharedRevealed = alreadyScanning.length
      ? Math.max(...alreadyScanning.map(s => s.scanState.revealed || 0))
      : 0;

    scout.scanState = {
      planet,
      total:60,
      remaining:sharedRemaining,
      revealed:sharedRevealed
    };
  }
}

function startScoutCooldown(scout){
  if(!scout || scout.type !== 'scout') return;
  scout.cooldownState = {remaining:120,total:120};

  if(scout.owner === 'player'){
    pushEvent(
      scout.name + ': исследование завершено. Начата зарядка солнечных батарей — 120 сек.',
      'discovery'
    );
  }
}

function updateScoutCooldown(scout, dtGame){
  if(!scout.cooldownState) return;
  scout.cooldownState.remaining = Math.max(0, scout.cooldownState.remaining - dtGame);

  if(scout.cooldownState.remaining <= 0){
    scout.cooldownState = null;
    if(scout.owner === 'player'){
      pushEvent(scout.name + ': батареи заряжены. Разведчик снова готов к полёту.', 'success');
    }else{
      scout.idleTimer = .5;
    }
  }
}

function updatePlanetScan(scout, dtGame){
  const scan = scout.scanState;
  if(!scan) return;

  const planet = scan.planet;

  // Если другой разведчик уже закончил сканирование этой планеты,
  // этот участник тоже завершает работу и уходит на перезарядку.
  if(scout.owner === 'player' && planet.playerScanned){
    scout.scanState = null;
    startScoutCooldown(scout);
    return;
  }

  const scanners = activePlanetScanners(planet, scout.owner);
  if(!scanners.length) return;

  // Сканирование общее для всех разведчиков у одной планеты.
  // Обновляет прогресс только первый активный разведчик, чтобы один кадр
  // не засчитывался несколько раз. Скорость равна числу разведчиков.
  const leader = scanners[0];
  if(leader !== scout) return;

  syncPlanetScanState(scanners);

  const speedMultiplier = scanners.length;
  const sharedScan = leader.scanState;
  sharedScan.remaining = Math.max(0, sharedScan.remaining - dtGame * speedMultiplier);

  scanners.forEach(s => {
    s.scanState.remaining = sharedScan.remaining;
  });

  const elapsed = sharedScan.total - sharedScan.remaining;

  if(scout.owner === 'player'){
    const resourceCount = planet.resources.length;
    let revealed = Math.max(
      sharedScan.revealed || 0,
      planet.playerDiscoveredResources.length
    );

    while(revealed < resourceCount){
      const threshold = 50 * (revealed + 1) / resourceCount;
      if(elapsed < threshold) break;

      const resource = planet.resources[revealed];
      if(!planet.playerDiscoveredResources.includes(resource)){
        planet.playerDiscoveredResources.push(resource);
        pushEvent(
          'Исследование «' + planet.name + '»: обнаружен ресурс — ' + resource + '.',
          'resource'
        );
      }
      revealed++;
    }

    scanners.forEach(s => {
      if(s.scanState) s.scanState.revealed = revealed;
    });
  }

  if(sharedScan.remaining <= 0){
    // Все разведчики, участвовавшие в совместном исследовании,
    // получают обычный двухминутный кулдаун после завершения.
    scanners.forEach(s => {
      s.scanState = null;
    });

    if(scout.owner === 'player'){
      completePlayerScan(scout, planet);
      scanners.forEach(startScoutCooldown);
    }else{
      enemyVisitedPlanetIds.add(planet.id);
      scanners.forEach(startScoutCooldown);
    }
  }
}

function fleetGroupMembers(group){
  if(!group) return [];
  const live = [];
  const ids = [];
  for(const id of group.shipIds || []){
    const ship = scouts.find(s => s.id === id && !s.destroyed && s.owner === 'player');
    if(ship){ live.push(ship); ids.push(id); }
  }
  group.shipIds = ids;
  return live;
}

function groupStatusText(group){
  const members = fleetGroupMembers(group);
  if(!members.length) return 'Нет кораблей';
  const combat = members.filter(s => s.combatTarget).length;
  if(combat) return 'В бою: ' + combat + ' из ' + members.length;
  if(members.every(s => s.patrolState)) return 'Вся группа патрулирует';

  const target = members[0].targetPlanet;
  if(target && members.every(s => s.targetPlanet === target)) return 'В пути к «' + target.name + '»';

  const orbit = members[0].orbitPlanet;
  if(orbit && members.every(s => s.orbitPlanet === orbit && !s.targetPlanet)) return 'На орбите «' + orbit.name + '»';

  if(members.every(s => s.stationary)) return 'Группа остановлена';
  const busy = members.filter(s => s.scanState || s.cooldownState || s.resourceTransport).length;
  if(busy) return 'Разные задачи • занято: ' + busy;
  return 'Разделена • разные приказы';
}
function createFleetGroup(ships){
  const valid = [...new Set((ships || []).filter(s => s && s.owner === 'player' && !s.destroyed))];
  if(valid.length < 2){
    pushEvent('Для группы флота выберите минимум два корабля.', 'discovery');
    return null;
  }
  if(fleetGroups.length >= 8){
    pushEvent('Можно иметь максимум 8 постоянных групп флота.', 'discovery');
    return null;
  }

  const ids = new Set(valid.map(s => s.id));
  for(const group of fleetGroups) group.shipIds = group.shipIds.filter(id => !ids.has(id));
  cleanupFleetGroups();

  const serialBefore = fleetGroupSerial;
  const group = {
    id:'fleet-group-' + Date.now() + '-' + Math.floor(Math.random()*10000),
    name:nextFleetGroupName(),
    colorIndex:serialBefore % 6,
    shipIds:valid.map(s => s.id)
  };
  fleetGroups.push(group);
  updateFleetGroupsBadge();
  pushEvent(fleetGroupLabel(group) + ' создана. Кораблей: ' + valid.length + '.', 'success');
  return group;
}

function canFleetTargetPlanet(fleet, planet){
  if(!fleet.length) return false;
  if(fleet.some(ship => ship.scanState || ship.cooldownState)) return false;

  if(planet.kind === 'researchAsteroid'){
    return isExploredPoint(planet.x,planet.y) && fleet.some(ship => ship.canMineResearch);
  }

  if(isPlanetDiscovered(planet)) return true;

  // В неизвестную систему весь отряд может идти за разведчиком,
  // если хотя бы один выбранный разведчик способен обнаружить цель.
  return fleet.some(ship => ship.canScan && canScoutTargetPlanet(ship,planet));
}

function sendSelectedFleetToPlanet(planet){
  const fleet = [...selectedFleet];
  if(!fleet.length) return;

  if(!canFleetTargetPlanet(fleet, planet)){
    scoutCommandHint.textContent =
      'Эта планета слишком далеко от зоны обнаружения выбранного флота.';
    scoutCommandHint.classList.add('show');
    clearTimeout(sendSelectedFleetToPlanet._timer);
    sendSelectedFleetToPlanet._timer = setTimeout(() => scoutCommandHint.classList.remove('show'), 2700);
    return;
  }

  const allAlreadyHere = fleet.every(ship => ship.orbitPlanet === planet && !ship.targetPlanet);
  if(allAlreadyHere){
    clearSelectedFleet();
    if(planet.kind === 'researchAsteroid'){
      scoutCommandHint.textContent = 'Выбранные корабли уже находятся у этого астероида.';
      scoutCommandHint.classList.add('show');
      setTimeout(() => scoutCommandHint.classList.remove('show'), 2000);
    }else{
      openPlanetInfo(planet);
    }
    return;
  }

  let eta = 0;
  fleet.forEach(ship => {
    const distance = Math.hypot(planet.x - ship.x, planet.y - ship.y);
    eta = Math.max(eta, Math.ceil(distance / Math.max(.1, ship.speed * gameSpeed)));
    setScoutTarget(ship, planet, {silent:true});
  });

  const count = fleet.length;
  clearSelectedFleet();
  clearScoutSelection();
  closePlanetInfo();

  const targetName = planet.kind === 'researchAsteroid'
    ? '«' + planet.name + '»'
    : (planet.playerScanned ? '«' + planet.name + '»' : 'неизвестной планете');
  scoutCommandHint.textContent =
    'Флот (' + count + ') отправлен к ' + targetName + '. В пути примерно ' + eta + ' сек.';
  scoutCommandHint.classList.add('show');
  clearTimeout(sendSelectedFleetToPlanet._timer2);
  sendSelectedFleetToPlanet._timer2 = setTimeout(() => scoutCommandHint.classList.remove('show'), 3600);

  pushEvent(
    'Флот из ' + count + ' корабл' + (count === 1 ? 'я' : 'ей') +
    ' отправлен к ' + targetName + '. Расчётное время перелёта: ' + eta + ' сек.',
    'discovery'
  );
}

function setScoutTarget(scout, planet, options={}){
  if(!scout || scout.owner !== 'player' && scout.owner !== 'enemy') return false;

  if(scout.resourceTransport && !options.preserveTransport){
    if(scout.owner === 'player' && !options.silent){
      scoutCommandHint.textContent =
        'Грузовой корабль выполняет доставку. Сначала завершите рейс или нажмите «Возврат».';
      scoutCommandHint.classList.add('show');
      setTimeout(()=>scoutCommandHint.classList.remove('show'),2600);
    }
    return false;
  }

  if(
    planet &&
    planet.kind === 'researchAsteroid' &&
    scout.owner === 'player' &&
    scout.canMineResearch
  ){
    if(scout.orbitPlanet && scout.orbitPlanet.kind !== 'researchAsteroid'){
      scout.homePlanet = scout.orbitPlanet;
    }
    if(getAsteroidMinerCount(planet, scout) >= 2){
      if(!options.silent){
        scoutCommandHint.textContent = 'На одном астероиде могут работать только 2 добывающих корабля.';
        scoutCommandHint.classList.add('show');
        setTimeout(() => scoutCommandHint.classList.remove('show'), 2600);
      }
      pushEvent('Для астероида уже назначено максимальное число добытчиков (2).', 'discovery');
      return false;
    }
  }

  if(
    planet &&
    planet.kind !== 'researchAsteroid' &&
    planet.owner === 'neutral' &&
    !planet.habitable &&
    scout.owner === 'player' &&
    scout.canMinePlanet
  ){
    if(!planet.playerScanned){
      if(!options.silent){
        scoutCommandHint.textContent = 'Сначала полностью исследуйте эту планету разведчиком.';
        scoutCommandHint.classList.add('show');
        setTimeout(() => scoutCommandHint.classList.remove('show'), 2600);
      }
      return false;
    }
    if(scout.orbitPlanet && scout.orbitPlanet.owner === 'player'){
      scout.homePlanet = scout.orbitPlanet;
    }
    if(getPlanetMinerCount(planet, scout) >= 2){
      if(!options.silent){
        scoutCommandHint.textContent = 'На этой планете уже работают 2 грузовых корабля.';
        scoutCommandHint.classList.add('show');
        setTimeout(() => scoutCommandHint.classList.remove('show'), 2600);
      }
      pushEvent('Для «' + planet.name + '» уже назначено максимальное число грузовых кораблей (2).', 'discovery');
      return false;
    }
  }

  if(scout.cooldownState){
    if(scout.owner === 'player'){
      scoutCommandHint.textContent = 'Разведчик заряжает солнечные батареи. Осталось ' + Math.ceil(scout.cooldownState.remaining) + ' сек.'; 
      scoutCommandHint.classList.add('show');
      setTimeout(() => scoutCommandHint.classList.remove('show'), 2400);
    }
    return;
  }

  if(scout.scanState){
    if(scout.owner === 'player'){
      scoutCommandHint.textContent = 'Разведчик ещё изучает текущую планету. Дождитесь окончания таймера.';
      scoutCommandHint.classList.add('show');
      setTimeout(() => scoutCommandHint.classList.remove('show'), 2200);
    }
    return;
  }

  if(scout.orbitPlanet === planet && !scout.targetPlanet) return;

  if(scout.owner === 'enemy' && scout.targetPlanet){
    enemyReservedPlanetIds.delete(scout.targetPlanet.id);
  }

  if(!options.preserveAnomaly) releaseShipSignalAssignment(scout);
  if(scout.orbitPlanet) scout.lastDepartureTarget = scout.orbitPlanet;
  scout.orbitPlanet = null;
  scout.targetPoint = null; clearWaypointMarker(scout);
  if(!options.preserveMining) scout.researchMining = null;
  if(!options.preservePlanetMining){
    scout.planetMining = null;
    scout.planetMiningCargo = {};
  }
  scout.stationary = false;
  scout.autoPatrolPlanet = null;
  scout.patrolState = null;
  scout.combatTarget = null;
  removePatrolVisual(scout);
  scout.targetPlanet = planet;

  if(scout.owner === 'enemy'){
    enemyReservedPlanetIds.add(planet.id);
  }else if(!options.silent){
    const distance = Math.hypot(planet.x - scout.x, planet.y - scout.y);
    const etaSeconds = Math.max(1, Math.ceil(distance / Math.max(0.1, scout.speed * gameSpeed)));

    clearSelectedFleet();
    clearScoutSelection();
    closePlanetFleetMenu();
    closePlanetInfo();

    const targetName = planet.kind === 'researchAsteroid'
      ? '«' + planet.name + '»'
      : (planet.playerScanned ? '«' + planet.name + '»' : 'неизвестной планете');
    scoutCommandHint.textContent =
      scout.name + ' отправлен к ' + targetName + '. В пути примерно ' + etaSeconds + ' сек.';
    scoutCommandHint.classList.add('show');
    clearTimeout(setScoutTarget._timer);
    setScoutTarget._timer = setTimeout(() => scoutCommandHint.classList.remove('show'), 3200);

    pushEvent(
      scout.name + ' отправлен к ' + targetName + '. Расчётное время полёта: ' + etaSeconds + ' сек.',
      'discovery'
    );
  }
  return true;
}

function chooseEnemyScoutTarget(scout){
  const candidates = planets
    .filter(p => p.owner === 'neutral')
    .filter(p => !enemyVisitedPlanetIds.has(p.id))
    .filter(p => !enemyReservedPlanetIds.has(p.id))
    .sort((a,b) => Math.hypot(a.x-scout.x,a.y-scout.y) - Math.hypot(b.x-scout.x,b.y-scout.y));

  if(candidates.length){
    setScoutTarget(scout, candidates[0]);
  }
}

const INFRA_UPGRADE_TYPES = new Set([
  'foodFarm','waterPlant','habitat','clinic','entertainment','securityCenter'
]);

const MAX_INFRA_UPGRADE = 3;      // +25%, +50%, +75%
const MAX_SCIENCE_LEVEL = 3;      // 10%, 20%, 30%
const MAX_MINING_LEVEL = 10;      // базовая станция = Ур.1, максимум Ур.10
const MINING_PRODUCTION_PER_LEVEL = .30; // +30% добычи за каждый новый уровень
const MINING_UPGRADE_COST_STEP = .25; // каждый следующий уровень добавляет +25% от базовой цены

function ensurePlanetUpgradeState(planet){
  if(!planet.upgrades){
    planet.upgrades = {
      foodFarm:0,waterPlant:0,habitat:0,clinic:0,entertainment:0,securityCenter:0,
      scienceCenter:0
    };
  }
  if(!planet.upgradeQueues) planet.upgradeQueues = {};
}

function isMiningBuildingType(type){
  const def = BUILDING_DEFS[type];
  return !!(def?.resource && def?.production);
}

function miningStationLevel(planet,type){
  if(!planet || (planet.buildings?.[type] || 0) <= 0) return 0;
  ensurePlanetUpgradeState(planet);
  return 1 + Math.max(0,planet.upgrades[type] || 0);
}

function miningProductionMultiplier(planet,type){
  const level = miningStationLevel(planet,type);
  if(level <= 0) return 0;
  return 1 + (level - 1) * MINING_PRODUCTION_PER_LEVEL;
}

function miningUpgradeCost(type,currentUpgradeCount){
  const def = BUILDING_DEFS[type];
  if(!def) return {};

  // Плавный линейный рост вместо экспоненциального:
  // Ур.1 → Ур.2 = ×1.25, Ур.3 = ×1.50 ... Ур.10 = ×3.25 от базовой цены.
  const factor = 1 + MINING_UPGRADE_COST_STEP * (currentUpgradeCount + 1);
  const result = {};
  for(const [resource,amount] of Object.entries(def.cost || {})){
    result[resource] = Math.max(
      5,
      Math.ceil((amount * factor) / 5) * 5
    );
  }
  return result;
}

function miningUpgradeDuration(type,currentUpgradeCount){
  const def = BUILDING_DEFS[type];
  const base = Math.max(20,def?.time || 30);
  return Math.ceil(base * (1.35 + currentUpgradeCount * .42));
}

function infrastructureMultiplier(planet,type){
  ensurePlanetUpgradeState(planet);
  const level = Math.max(0,planet.upgrades[type] || 0);
  return 1 + level * .25;
}

function infrastructureBonusText(planet,type){
  const level = planet.upgrades?.[type] || 0;
  return '+' + (level * 25) + '%';
}

function scaledUpgradeCost(type,currentLevel){
  const def = BUILDING_DEFS[type];
  if(!def) return {};
  const factors = [0.70,1.25,2.05];
  const factor = factors[Math.min(currentLevel,factors.length-1)];
  const result = {};
  for(const [resource,amount] of Object.entries(def.cost || {})){
    result[resource] = Math.max(5,Math.ceil(amount * factor / 5) * 5);
  }
  return result;
}

function upgradeDuration(type,currentLevel){
  if(type === 'scienceCenter') return [65,95][Math.min(currentLevel,1)];
  return [40,65,95][Math.min(currentLevel,2)];
}

function scienceCenterLevel(planet){
  if(!planet || (planet.buildings?.scienceCenter || 0) <= 0) return 0;
  ensurePlanetUpgradeState(planet);
  return 1 + (planet.upgrades.scienceCenter || 0);
}

function scienceCenterBonus(planet){
  return scienceCenterLevel(planet) * .10 + civilizationScienceBonus(planet);
}

function globalScienceBonus(){
  let bonus = 0;
  planets.forEach(p => {
    if(p.owner === 'player') bonus += scienceCenterBonus(p);
  });
  return Math.min(1,bonus); // максимум +100%
}

function researchSpeedMultiplier(){
  return 1 + globalScienceBonus();
}

function scienceUpgradeCost(planet){
  const level = scienceCenterLevel(planet);
  if(level === 1) return {'Железо':220,'Медь':120,'Литий':60,'Алюминий':35};
  if(level === 2) return {'Железо':360,'Медь':180,'Литий':90,'Алюминий':60};
  return {};
}

function startInfrastructureUpgrade(planet,type){
  if(!planet || planet.owner !== 'player') return;
  ensurePlanetUpgradeState(planet);

  const buildingCount = planet.buildings?.[type] || 0;
  if(buildingCount <= 0){
    pushEvent('Сначала постройте «' + (BUILDING_DEFS[type]?.name || type) + '».', 'discovery');
    return;
  }

  if(planet.upgradeQueues[type]) return;

  const current = planet.upgrades[type] || 0;
  const max = type === 'scienceCenter' ? 2 : MAX_INFRA_UPGRADE;
  if(current >= max) return;

  const cost = type === 'scienceCenter'
    ? scienceUpgradeCost(planet)
    : scaledUpgradeCost(type,current);

  if(!spendCost(planet,cost)){
    pushEvent('Недостаточно ресурсов для улучшения «' + BUILDING_DEFS[type].name + '».', 'discovery');
    return;
  }

  const duration = upgradeDuration(type,current);
  planet.upgradeQueues[type] = {
    type,
    cost:{...cost},
    remaining:duration,
    total:duration
  };

  pushEvent('Начато улучшение «' + BUILDING_DEFS[type].name + '» на «' + planet.name + '».', 'discovery');
  if(currentFleetPlanet === planet) renderPlanetMenuTab(planet);
}

function cancelInfrastructureUpgrade(planet,type){
  if(!planet?.upgradeQueues?.[type]) return;
  const q = planet.upgradeQueues[type];
  refundCost(planet,q.cost || {});
  delete planet.upgradeQueues[type];
  pushEvent('Улучшение «' + BUILDING_DEFS[type].name + '» отменено. Ресурсы возвращены.', 'discovery');
  if(currentFleetPlanet === planet) renderPlanetMenuTab(planet);
}

function infrastructureUpgradeCardHtml(planet,type){
  ensurePlanetUpgradeState(planet);
  const def = BUILDING_DEFS[type];

function startMiningUpgrade(planet,type){
  if(!planet || planet.owner !== 'player' || !isMiningBuildingType(type)) return;
  ensurePlanetUpgradeState(planet);

  if((planet.buildings?.[type] || 0) <= 0){
    pushEvent('Сначала постройте «' + BUILDING_DEFS[type].name + '».', 'discovery');
    return;
  }
  if(planet.upgradeQueues[type]) return;

  const currentUpgradeCount = Math.max(0,planet.upgrades[type] || 0);
  const currentLevel = 1 + currentUpgradeCount;
  if(currentLevel >= MAX_MINING_LEVEL) return;

  const cost = miningUpgradeCost(type,currentUpgradeCount);
  if(!spendCost(planet,cost)){
    pushEvent(
      'Недостаточно ресурсов для улучшения «' + BUILDING_DEFS[type].name +
      '» до уровня ' + (currentLevel + 1) + '.',
      'discovery'
    );
    return;
  }

  const duration = miningUpgradeDuration(type,currentUpgradeCount);
  planet.upgradeQueues[type] = {
    type,
    kind:'mining',
    cost:{...cost},
    remaining:duration,
    total:duration
  };

  pushEvent(
    'Начато улучшение «' + BUILDING_DEFS[type].name + '»: Ур. ' +
    currentLevel + ' → Ур. ' + (currentLevel + 1) + '.',
    'discovery'
  );

  updateSelectedPlanetResourceBar(true);
  if(currentFleetPlanet === planet) renderPlanetMenuTab(planet);
}

}

const POPULATION_NEEDS = {
  food:{label:'Еда', icon:'🍽', building:'foodFarm', weight:1.35},
  water:{label:'Вода', icon:'💧', building:'waterPlant', weight:1.45},
  housing:{label:'Жильё', icon:'⌂', building:'habitat', weight:1.20},
  medicine:{label:'Медицина', icon:'✚', building:'clinic', weight:.85},
  entertainment:{label:'Развлечения', icon:'★', building:'entertainment', weight:.70},
  security:{label:'Безопасность', icon:'◆', building:'securityCenter', weight:.80}
};

function populationNeedData(planet){
  const population = Math.max(1, planet.population || 0);
  const result = {};
  for(const [key,need] of Object.entries(POPULATION_NEEDS)){
    const def = BUILDING_DEFS[need.building];
    const count = planet.buildings?.[need.building] || 0;
    const capacity = count * (def.support || 0) * infrastructureMultiplier(planet,need.building);
    const ratio = population <= 0 ? 1 : Math.min(1.15, capacity / population);
    result[key] = {
      ...need,
      count,
      capacity,
      ratio,
      percent:Math.round(Math.min(100,ratio*100))
    };
  }
  return result;
}

function calculateSatisfaction(planet){
  if(planet.owner !== 'player' && planet.owner !== 'enemy') return 0;
  if((planet.population || 0) <= 0) return 0;

  const needs = populationNeedData(planet);
  let weighted = 0, totalWeight = 0;
  for(const need of Object.values(needs)){
    weighted += Math.min(1,need.ratio) * need.weight;
    totalWeight += need.weight;
  }

  let score = totalWeight ? (weighted / totalWeight) * 100 : 0;

  // Severe shortages of food/water are felt more strongly.
  if(needs.food.ratio < .55) score -= (0.55 - needs.food.ratio) * 30;
  if(needs.water.ratio < .55) score -= (0.55 - needs.water.ratio) * 35;
  if(needs.housing.ratio < .65) score -= (0.65 - needs.housing.ratio) * 18;

  return Math.max(0,Math.min(100,Math.round(score)));
}

function populationEfficiency(planet){
  const s = calculateSatisfaction(planet);
  if(s >= 80) return 1.05;
  if(s >= 60) return 1;
  if(s >= 40) return .90;
  if(s >= 20) return .75;
  return .55;
}

function populationGrowthPerMinute(planet){
  const s = calculateSatisfaction(planet);
  if(s >= 90) return .0060;   // +0.60% / min
  if(s >= 80) return .0045;
  if(s >= 60) return .0025;
  if(s >= 40) return .0004;
  if(s >= 20) return -.0015;
  return -.0040;
}

function populationTrendText(planet){
  const rate = populationGrowthPerMinute(planet);
  if(rate > .003) return 'Быстрый рост';
  if(rate > .0005) return 'Рост';
  if(rate >= 0) return 'Стабильно';
  if(rate > -.0025) return 'Снижение';
  return 'Сильный отток';
}

function updatePopulation(planet, dtGame){
  if(planet.owner !== 'player' || !planet.habitable || planet.population <= 0) return;

  const old = planet.population;
  const ratePerMinute = populationGrowthPerMinute(planet);
  planet.population = Math.max(0, planet.population + planet.population * ratePerMinute * (dtGame/60));

  // Avoid spamming: notify only on large threshold changes.
  const oldBand = Math.floor(old / 250);
  const newBand = Math.floor(planet.population / 250);
  if(oldBand !== newBand && Math.abs(newBand-oldBand) >= 1){
    pushEvent(
      'Население «' + planet.name + '»: ' + Math.floor(planet.population).toLocaleString('ru-RU') +
      ' • ' + populationTrendText(planet) + '.',
      ratePerMinute >= 0 ? 'success' : 'alert'
    );
  }
}

function getProductionRate(planet, resource){
  let total = 0;
  for(const [type,def] of Object.entries(BUILDING_DEFS)){
    if(def.resource === resource && def.production){
      const count = planet.buildings?.[type] || 0;
      if(count > 0){
        total += count *
          def.production *
          miningProductionMultiplier(planet,type) *
          civilizationProductionMultiplier(planet);
      }
    }
  }
  return total;
}

function localStock(planet, resource){
  return Number(planet?.stockpile?.[resource] || 0);
}
function empireStock(resource){
  let total = 0;
  for(const colony of planets){
    if(colony.owner === 'player') total += localStock(colony,resource);
  }
  return total;
}
function empireResourceKeys(){
  const keys = new Set();
  for(const colony of planets){
    if(colony.owner !== 'player') continue;
    Object.keys(colony.stockpile || {}).forEach(r=>keys.add(r));
    (colony.resources || []).forEach(r=>keys.add(r));
  }
  return [...keys];
}
function empireProductionRate(resource){
  return playerColonies().reduce((sum,colony)=>sum + getProductionRate(colony,resource),0);
}
function getStock(planet, resource){
  // Все собственные колонии пользуются единым имперским резервом.
  if(planet?.owner === 'player') return empireStock(resource);
  return localStock(planet,resource);
}
function withdrawEmpireStock(resource, amount, preferredPlanet=null){
  let remaining = Math.max(0,Number(amount || 0));
  const ordered = playerColonies().slice();
  if(preferredPlanet?.owner === 'player'){
    ordered.sort((a,b)=>(a===preferredPlanet?-1:b===preferredPlanet?1:0));
  }
  for(const colony of ordered){
    if(remaining <= 0) break;
    if(!colony.stockpile) colony.stockpile = {};
    const available = Math.max(0,localStock(colony,resource));
    const take = Math.min(available,remaining);
    if(take > 0){
      colony.stockpile[resource] = available - take;
      remaining -= take;
    }
  }
  return remaining <= 0.000001;
}
function addStock(planet, resource, amount){
  if(!planet) return;
  if(!planet.stockpile) planet.stockpile = {};
  const delta = Number(amount || 0);
  if(planet.owner === 'player' && delta < 0){
    withdrawEmpireStock(resource,-delta,planet);
    return;
  }
  planet.stockpile[resource] = localStock(planet,resource) + delta;
}
function canAfford(planet, cost){
  return Object.entries(cost).every(([res,amount]) => getStock(planet,res) >= amount);
}
function spendCost(planet, cost){
  if(!canAfford(planet,cost)) return false;
  Object.entries(cost).forEach(([res,amount]) => addStock(planet,res,-amount));
  return true;
}
function formatCost(cost){
  return Object.entries(cost).map(([r,a]) => `${r}: ${a}`).join(' • ');
}
function stockpileHtml(planet){
  const important = ['Железо','Медь','Литий','Никель','Титан','Платина','Серебро'];
  const sourceKeys = planet?.owner === 'player' ? empireResourceKeys() : Object.keys(planet.stockpile || {});
  const keys = [...new Set([...sourceKeys, ...important])]
    .filter(k => getStock(planet,k) > 0 || ['Железо','Медь','Литий'].includes(k));

  const note = planet?.owner === 'player'
    ? '<div class="empireStockNote">Общий резерв империи • ресурсы всех ваших колоний доступны здесь без ручной перевозки.</div>'
    : '';
  return note + `<div class="stockpileGrid">${
    keys.map(r => {
      const rate = planet?.owner === 'player' ? empireProductionRate(r) : getProductionRate(planet,r);
      const prod = rate > 0
        ? `<div class="stockProd">+${rate.toFixed(2)}/с</div>`
        : '';
      return `<div class="stockItem"><div class="stockName">${resourceIconHtml(r,'sm')}<span>${r}</span></div><div class="stockValue">${Math.floor(getStock(planet,r))}</div>${prod}</div>`;
    }).join('')
  }</div>`;
}

function removeOneShipOrder(planet,type){
  const q = planet.shipQueue;
  if(!q) return false;
  const def = SHIP_DEFS[type];
  if(!def) return false;

  if(Array.isArray(q.pending)){
    for(let i=q.pending.length-1;i>=0;i--){
      if(q.pending[i] === type){
        q.pending.splice(i,1);
        refundCost(planet,def.cost);
        pushEvent('Из очереди удалён корабль «' + def.name + '». Ресурсы возвращены.', 'discovery');
        if(currentFleetPlanet === planet) renderPlanetMenuTab(planet);
        return true;
      }
    }
  }

  if(q.type === type){
    const pending = Array.isArray(q.pending) ? q.pending : [];
    const nextType = pending.length ? pending.shift() : null;
    refundCost(planet,def.cost);
    if(nextType){
      const nextDef = SHIP_DEFS[nextType];
      planet.shipQueue = {type:nextType,remaining:nextDef.time,total:nextDef.time,pending};
    }else{
      planet.shipQueue = null;
    }
    pushEvent('Строительство корабля «' + def.name + '» отменено. Ресурсы возвращены.', 'discovery');
    if(currentFleetPlanet === planet) renderPlanetMenuTab(planet);
    return true;
  }
  return false;
}

function techKey(name){
  return String(name).toLowerCase().replace(/[^a-zа-яё0-9]+/gi,'-').replace(/^-|-$/g,'');
}
function researchDuration(level){ return 20 + level * 20; }
function startTechnology(name,level,cost){
  const key = techKey(name);
  if(completedTechs.has(key)) return;
  if(researchQueue){
    pushEvent('Сначала завершите или отмените текущее исследование.', 'discovery');
    return;
  }
  if(playerKsenit < cost){
    pushEvent('Недостаточно Ксенита для исследования «' + name + '».', 'discovery');
    return;
  }
  playerKsenit -= cost;
  const duration = researchDuration(level);
  researchQueue = {key,name,level,cost,total:duration,remaining:duration};
  pushEvent('Начато исследование «' + name + '».', 'discovery');
  if(currentPlanetTab === 'research' && currentFleetPlanet) renderResearchTab(currentFleetPlanet);
}
function cancelTechnology(){
  if(!researchQueue) return;
  playerKsenit += researchQueue.cost;
  pushEvent('Исследование «' + researchQueue.name + '» отменено. Ксенит возвращён.', 'discovery');
  researchQueue = null;
  if(currentPlanetTab === 'research' && currentFleetPlanet) renderResearchTab(currentFleetPlanet);
}
function updateResearch(dtGame){
  if(!researchQueue) return;
  researchQueue.remaining -= dtGame * researchSpeedMultiplier();
  if(researchQueue.remaining <= 0){
    const done = researchQueue;
    completedTechs.add(done.key);
    researchQueue = null;
    pushEvent('Исследование «' + done.name + '» завершено.', 'success');
    if(currentPlanetTab === 'research' && currentFleetPlanet) renderResearchTab(currentFleetPlanet);
  }
}

function startBuilding(planet, type){
  const def = BUILDING_DEFS[type];
  if(!def || planet.owner !== 'player') return;

  const requiredCivilizationLevel = requiredCivilizationLevelForBuilding(type);
  if(getCivilizationLevel(planet) < requiredCivilizationLevel){
    const info = civilizationInfo(requiredCivilizationLevel);
    pushEvent(
      '«' + def.name + '» откроется на уровне цивилизации ' +
      requiredCivilizationLevel + ' — «' + info.name + '».',
      'discovery'
    );
    return;
  }

  if(!planet.buildingQueues) planet.buildingQueues = {};
  if(planet.buildingQueues[type]) return;

  if(def.single && (planet.buildings?.[type] || 0) >= 1){
    pushEvent('На этой планете уже построен «' + def.name + '». Его можно улучшать.', 'discovery');
    return;
  }

  if(def.resource && !planet.resources.includes(def.resource)){
    pushEvent('На этой планете нет месторождения ресурса «' + def.resource + '».', 'discovery');
    return;
  }

  if(!spendCost(planet,def.cost)){
    pushEvent('Недостаточно ресурсов для постройки: ' + def.name + '.', 'discovery');
    return;
  }

  planet.buildingQueues[type] = {
    type,
    remaining:def.time,
    total:def.time
  };

  pushEvent('Начато строительство: ' + def.name + ' на «' + planet.name + '».', 'discovery');

  updateSelectedPlanetResourceBar();
  if(currentFleetPlanet === planet) renderPlanetMenuTab(planet);
}

function getQueuedShipCount(planet, type){
  const q = planet.shipQueue;
  if(!q) return 0;
  let count = q.type === type ? 1 : 0;
  if(Array.isArray(q.pending)){
    q.pending.forEach(t => { if(t === type) count++; });
  }
  return count;
}

function startShipBuild(planet, type){
  const def = SHIP_DEFS[type];
  if(!def || planet.owner !== 'player') return;

  const requiredCivilizationLevel = requiredCivilizationLevelForShip(type);
  if(getCivilizationLevel(planet) < requiredCivilizationLevel){
    const info = civilizationInfo(requiredCivilizationLevel);
    pushEvent(
      'Корабль «' + def.name + '» откроется на уровне цивилизации ' +
      requiredCivilizationLevel + ' — «' + info.name + '».',
      'discovery'
    );
    return;
  }

  if(type === 'colonizer' && (planet.population || 0) < 500){
    pushEvent(
      'Для колонизационного корабля на «' + planet.name + '» нужно минимум 500 жителей. На борт отправятся 300 колонистов.',
      'discovery'
    );
    return;
  }

  if(!spendCost(planet,def.cost)){
    pushEvent('Недостаточно ресурсов для корабля «' + def.name + '».', 'discovery');
    return;
  }

  if(!planet.shipQueue){
    planet.shipQueue = {type,remaining:def.time,total:def.time,pending:[]};
  }else{
    if(!Array.isArray(planet.shipQueue.pending)) planet.shipQueue.pending = [];
    planet.shipQueue.pending.push(type);
  }

  const queuedCount = getQueuedShipCount(planet, type);
  pushEvent(
    'На верфи «' + planet.name + '» в очередь добавлен корабль «' + def.name + '» × ' + queuedCount + '.',
    'discovery'
  );

}

function updateEconomy(dtGame){
  planets.forEach(p => {
    if(p.owner !== 'player') return;

    // Производство всех добывающих сооружений.
    for(const [type,def] of Object.entries(BUILDING_DEFS)){
      if(def.resource && def.production){
        const count = p.buildings?.[type] || 0;
        if(count > 0){
          addStock(
            p,
            def.resource,
            count *
            def.production *
            miningProductionMultiplier(p,type) *
            civilizationProductionMultiplier(p) *
            populationEfficiency(p) *
            dtGame
          );
        }
      }
    }

    updatePopulation(p,dtGame);

    ensurePlanetUpgradeState(p);

    for(const [type,queue] of Object.entries({...p.upgradeQueues})){
      queue.remaining -= dtGame;

      if(queue.remaining <= 0){
        delete p.upgradeQueues[type];
        p.upgrades[type] = (p.upgrades[type] || 0) + 1;

        const isScience = type === 'scienceCenter';
        const isMining = isMiningBuildingType(type);
        const value = isScience
          ? '+' + scienceCenterLevel(p)*10 + '% к скорости исследований'
          : isMining
            ? 'уровень ' + miningStationLevel(p,type) +
              ' • добыча +' + (
                BUILDING_DEFS[type].production *
                miningProductionMultiplier(p,type) *
                civilizationProductionMultiplier(p)
              ).toFixed(2) +
              ' ' + BUILDING_DEFS[type].resource + '/с'
            : '+' + (p.upgrades[type]*25) + '% к обеспечению';

        pushEvent(
          'Улучшение «' + BUILDING_DEFS[type].name + '» завершено на «' + p.name + '»: ' + value + '.',
          'success'
        );

        if(resourceBarPlanet === p) updateSelectedPlanetResourceBar(true);

        if(currentFleetPlanet === p){
          renderPlanetMenuTab(p);
        }
      }
    }

    if(!p.buildingQueues) p.buildingQueues = {};
    for(const [type,queue] of Object.entries({...p.buildingQueues})){
      queue.remaining -= dtGame;

      if(queue.remaining <= 0){
        delete p.buildingQueues[type];
        p.buildings[type] = (p.buildings[type] || 0) + 1;
        pushEvent(BUILDING_DEFS[type].name + ' построена на «' + p.name + '».', 'success');

        if(resourceBarPlanet === p) updateSelectedPlanetResourceBar();
      }
    }

    if(p.shipQueue){
      p.shipQueue.remaining -= dtGame;
      if(p.shipQueue.remaining <= 0){
        const finishedType = p.shipQueue.type;
        const pending = Array.isArray(p.shipQueue.pending) ? p.shipQueue.pending : [];
        const nextType = pending.length ? pending.shift() : null;
        const ship = makeBuiltShip('player',p,finishedType);

        const autoPatrolEnabled = isMilitaryShipType(finishedType)
          ? enablePatrol(ship,{
              silent:true,
              aroundPlanet:p,
              radius:175,
              autoAssigned:true
            })
          : false;

        pushEvent(
          'Построен корабль «' + ship.name + '» на «' + p.name + '». ' +
          (autoPatrolEnabled
            ? 'Военный корабль автоматически переведён в патруль вокруг планеты.'
            : ''),
          'success'
        );

        if(nextType){
          const nextDef = SHIP_DEFS[nextType];
          p.shipQueue = {
            type:nextType,
            remaining:nextDef.time,
            total:nextDef.time,
            pending
          };
        }else{
          p.shipQueue = null;
        }

        if(currentFleetPlanet === p) renderPlanetMenuTab(p);
      }
    }

    updateCivilizationProgress(p);
  });
}

  if(!mining) return;

  const asteroid = mining.asteroid;
  if(asteroid.reserve <= 0){
    ship.researchMining = null;
    ship.miningCargo = 0;
    pushEvent('Запасы Ксенита на астероиде исчерпаны.', 'discovery');
    return;
  }

  if(mining.stage === 'mining'){
    mining.remaining = Math.max(0, mining.remaining - dtGame);

      mining.stage = 'returning';

      setScoutTarget(ship, mining.homePlanet, {silent:true,preserveMining:true});
    }
  }
}


function ensurePlanetExtractionReserves(planet){
  if(planet.extractionReserves) return planet.extractionReserves;
  planet.extractionReserves = {};
  const resources = [...new Set((planet.resources || []).filter(Boolean))];
  if(!resources.length) return planet.extractionReserves;

  // Каждая непригодная для колонизации планета содержит ровно 10 000 ед.
  // местных ресурсов. Пропорции случайны, поэтому ценность таких миров различается.
  const TOTAL = 10000;
  const minShare = resources.length > 1 ? Math.min(500, Math.floor(TOTAL / (resources.length * 3))) : 0;
  const guaranteed = minShare * resources.length;
  const distributable = TOTAL - guaranteed;
  const weights = resources.map(() => .35 + Math.random() * 1.65);
  const weightSum = weights.reduce((a,b)=>a+b,0) || 1;
  let assigned = 0;

  resources.forEach((resource,index) => {
    const variable = index === resources.length - 1
      ? distributable - assigned
      : Math.floor(distributable * weights[index] / weightSum);
    assigned += variable;
    planet.extractionReserves[resource] = minShare + variable;
  });

  // Защита от округления: итог всегда должен быть ровно 10 000.
  const actual = Object.values(planet.extractionReserves).reduce((sum,v)=>sum+v,0);
  if(actual !== TOTAL){
    planet.extractionReserves[resources[0]] += TOTAL - actual;
  }
  planet.extractionReserveTotal = TOTAL;
  return planet.extractionReserves;
}

function planetExtractionRemaining(planet){
  const reserves = ensurePlanetExtractionReserves(planet);
  return Object.values(reserves).reduce((sum,v)=>sum + Math.max(0,v || 0),0);
}

function getPlanetMinerCount(planet, ignoreShip=null){
  return scouts.filter(ship =>
    ship !== ignoreShip &&
    ship.owner === 'player' &&
    ship.canMinePlanet &&
    (
      (ship.targetPlanet && ship.targetPlanet.id === planet.id) ||
      (ship.planetMining && ship.planetMining.planet && ship.planetMining.planet.id === planet.id)
    )
  ).length;
}

function startPlanetResourceMining(ship, planet){
  if(!ship || !ship.canMinePlanet || !planet || planet.habitable || planet.owner !== 'neutral') return false;
  if(planetExtractionRemaining(planet) <= 0){
    ship.planetMining = null;
    ship.planetMiningCargo = {};
    pushEvent('Месторождения на «' + planet.name + '» полностью исчерпаны. ' + ship.name + ' остаётся на орбите и ждёт приказа.', 'discovery');
    return false;
  }
  const homePlanet = ship.homePlanet || ship.lastDepartureTarget || playerPlanet;
  ship.planetMining = {
    planet,
    homePlanet,
    stage:'mining',
    remaining:18,
    miningTime:18,
    capacity:ship.cargoCapacity || ship.planetMiningCapacity || 18,
    totalDelivered:0,
    loopActive:true
  };
  ship.planetMiningCargo = {};
  pushEvent(ship.name + ' начал загрузку ресурсов на «' + planet.name + '».', 'resource');
  return true;
}

function loadPlanetMiningCargo(ship){
  const mining = ship.planetMining;
  if(!mining) return 0;
  const reserves = ensurePlanetExtractionReserves(mining.planet);
  const active = Object.keys(reserves).filter(r => reserves[r] > 0);
  if(!active.length) return 0;

  let remaining = Math.max(1, Math.floor(mining.capacity || 18));
  const cargo = {};
  while(remaining > 0){
    const available = active.filter(r => reserves[r] > 0);
    if(!available.length) break;
    let changed = false;
    for(const resource of available){
      if(remaining <= 0) break;
      const take = Math.min(reserves[resource], Math.max(1, Math.floor(remaining / available.length)));
      if(take <= 0) continue;
      reserves[resource] -= take;
      cargo[resource] = (cargo[resource] || 0) + take;
      remaining -= take;
      changed = true;
    }
    if(!changed) break;
  }
  ship.planetMiningCargo = cargo;
  return Object.values(cargo).reduce((sum,v)=>sum+v,0);
}

function depositPlanetMiningCargo(ship, homePlanet){
  const mining = ship.planetMining;
  if(!mining) return;
  const cargo = ship.planetMiningCargo || {};
  let delivered = 0;
  const parts = [];
  for(const [resource,amount] of Object.entries(cargo)){
    if(amount <= 0) continue;
    addStock(homePlanet,resource,amount);
    delivered += amount;
    parts.push(resource + ' +' + amount);
  }
  mining.totalDelivered = (mining.totalDelivered || 0) + delivered;
  ship.planetMiningCargo = {};

  if(delivered > 0){
    pushEvent(ship.name + ' доставил на «' + homePlanet.name + '»: ' + parts.join(' • ') + '.', 'resource');
    if(resourceBarPlanet === homePlanet) updateSelectedPlanetResourceBar(true);
  }

  if(mining.loopActive && planetExtractionRemaining(mining.planet) > 0){
    mining.stage = 'toPlanet';
    setScoutTarget(ship,mining.planet,{silent:true,preservePlanetMining:true});
    pushEvent(ship.name + ' выгрузил ресурсы и снова направился к «' + mining.planet.name + '».', 'discovery');
  }else{
    if(planetExtractionRemaining(mining.planet) <= 0){
      pushEvent('Месторождения на «' + mining.planet.name + '» исчерпаны. ' + ship.name + ' завершил рейсы.', 'discovery');
    }else{
      pushEvent(ship.name + ' отозван с автономной добычи и остаётся на базе.', 'discovery');
    }
    ship.planetMining = null;
  }
}

function updatePlanetResourceMining(ship,dtGame){
  const mining = ship.planetMining;
  if(!mining || mining.stage !== 'mining') return;
  mining.remaining = Math.max(0,mining.remaining-dtGame);
  if(mining.remaining > 0) return;

  const load = loadPlanetMiningCargo(ship);
  if(load <= 0){
    const exhaustedPlanet = mining.planet;
    ship.planetMining = null;
    ship.planetMiningCargo = {};
    ship.stationary = false;
    // Корабль уже находится на орбите этой планеты: остаётся здесь и ждёт приказа.
    pushEvent('На «' + exhaustedPlanet.name + '» больше нет ресурсов. ' + ship.name + ' остаётся на орбите и ждёт дальнейших указаний.', 'discovery');
    if(ship.owner === 'player'){
      scoutCommandHint.textContent = ship.name + ': месторождение исчерпано. Корабль остаётся на орбите «' + exhaustedPlanet.name + '».';
      scoutCommandHint.classList.add('show');
      clearTimeout(updatePlanetResourceMining._hintTimer);
      updatePlanetResourceMining._hintTimer = setTimeout(()=>scoutCommandHint.classList.remove('show'),3200);
    }
    return;
  }

  mining.stage = 'returning';
  pushEvent(ship.name + ' загрузил ' + load + ' ед. ресурсов на «' + mining.planet.name + '» и возвращается на базу.', 'resource');
  setScoutTarget(ship,mining.homePlanet,{silent:true,preservePlanetMining:true});
}

function updateCaptures(dtGame){
  planets.forEach(p => {
    const c = p.captureState;
    if(!c) return;

    c.remaining = Math.max(0,c.remaining-dtGame);
    const progress = 1 - c.remaining/c.total;
    p.population = Math.max(0, Math.floor(c.startPopulation * (1 - .30*progress)));

    if(c.remaining <= 0){
      p.owner = c.attackerOwner;
      p.buildings = {
        ironMine:0,copperMine:0,lithiumMine:0,
        foodFarm:0,waterPlant:0,habitat:0,clinic:0,entertainment:0,securityCenter:0,
        scienceCenter:0,
        missileBattery:0,laserBattery:0,shipyard:0
      };
      p.upgrades = {
        foodFarm:0,waterPlant:0,habitat:0,clinic:0,entertainment:0,securityCenter:0,
        scienceCenter:0
      };
      p.upgradeQueues = {};
      p.buildingQueues = {};
      p.buildingQueue = null;
      p.shipQueue = null;
      p.captureState = null;
      updatePlanetMapData(p);
      pushEvent('«' + p.name + '» захвачена. Все постройки уничтожены.', 'discovery');
    }
  });
}

function getBuildingIconMeta(type){
  const meta = {
    ironMine:{glyph:'Fe', orb:'iron', mode:'mine'},
    copperMine:{glyph:'Cu', orb:'copper', mode:'mine'},
    lithiumMine:{glyph:'Li', orb:'lithium', mode:'mine'},
    foodFarm:{glyph:'🌿', orb:'food', mode:'emoji'},
    waterPlant:{glyph:'💧', orb:'water', mode:'emoji'},
    habitat:{glyph:'🏠', orb:'housing', mode:'emoji'},
    clinic:{glyph:'✚', orb:'medicine', mode:'emoji'},
    entertainment:{glyph:'✦', orb:'entertainment', mode:'emoji'},
    securityCenter:{glyph:'🛡', orb:'security', mode:'emoji'},
    scienceCenter:{glyph:'⚛', orb:'science', mode:'emoji'},
    missileBattery:{glyph:'🚀', orb:'missile', mode:'emoji'},
    laserBattery:{glyph:'✹', orb:'laser', mode:'emoji'}
  };
  if(meta[type]) return meta[type];
  const def = BUILDING_DEFS[type];
  if(def?.resource){
    const res = resourceVisualMeta(def.resource);
    return {glyph:res.short || def.resource.slice(0,2), orb:res.cls || 'iron', mode:'mine'};
  }
  return {glyph:'⚙', orb:'iron', mode:'emoji'};
}

function buildingCardHtml(planet,type){
  const def = BUILDING_DEFS[type];
  const requiredCivilizationLevel = requiredCivilizationLevelForBuilding(type);

  if(getCivilizationLevel(planet) < requiredCivilizationLevel){
    return `<div class="buildCard ${type} civilizationLockedCard">
      <div class="buildHead">
        ${buildingIconHtml(type)}
        <div class="buildText">
          <div class="buildName">${def.name}</div>
          <div class="buildDesc">${def.desc}</div>
        </div>
      </div>
      ${lockedCivilizationHtml(requiredCivilizationLevel)}
    </div>`;
  }

  if(!planet.buildingQueues) planet.buildingQueues = {};
  const queue = planet.buildingQueues[type];
  const inQueue = !!queue;
  const affordable = canAfford(planet,def.cost);
  const unavailableDeposit = !!def.resource && !planet.resources.includes(def.resource);
  const count = planet.buildings[type] || 0;
  const singleBuilt = !!def.single && count >= 1;
  const disabled = inQueue || !affordable || unavailableDeposit || singleBuilt;

  let productionInfo = '';
  let buildingLabel = def.name + (count ? ' • ' + count : '');

  if(def.resource && def.production){
    const level = miningStationLevel(planet,type);
    const rate = count > 0
      ? def.production *
        miningProductionMultiplier(planet,type) *
        civilizationProductionMultiplier(planet)
      : def.production;

    productionInfo = `<div class="buildTime">
      ${count > 0 ? 'Текущая добыча' : 'Базовая добыча'}:
      +${rate.toFixed(2)} ${def.resource}/с
    </div>`;

    buildingLabel = def.name + (count ? ' • Ур. ' + level : '');
  }

  // Добывающая станция после постройки отображается одной карточкой:
  // текущий уровень, добыча, цена следующего уровня и кнопка улучшения — всё вместе.
  if(singleBuilt && !inQueue && isMiningBuildingType(type)){
    ensurePlanetUpgradeState(planet);

    const currentUpgradeCount = Math.max(0,planet.upgrades[type] || 0);
    const currentLevel = 1 + currentUpgradeCount;
    const atMax = currentLevel >= MAX_MINING_LEVEL;
    const nextLevel = Math.min(MAX_MINING_LEVEL,currentLevel + 1);
    const upgradeQueue = planet.upgradeQueues[type];

    const civMult = civilizationProductionMultiplier(planet);
    const currentRate = def.production * miningProductionMultiplier(planet,type) * civMult;
    const nextMultiplier = 1 + (nextLevel - 1) * MINING_PRODUCTION_PER_LEVEL;
    const nextRate = def.production * nextMultiplier * civMult;
    const upgradeCost = atMax ? {} : miningUpgradeCost(type,currentUpgradeCount);

    return `<div class="buildCard ${type} builtSingleStation combinedMineCard">
      <div class="buildHead">
        ${buildingIconHtml(type)}
        <div class="buildText">
          <div class="buildName">${def.name}</div>
          <div class="buildDesc">${def.desc}</div>
        </div>
        <div class="mineLevelPill">УР. ${currentLevel}</div>
      </div>

      <div class="combinedMineCurrent">
        <div>
          <span class="combinedMineLabel">Текущая добыча</span>
          <b>+${currentRate.toFixed(2)} ${def.resource}/с</b>
        </div>
        <div class="combinedMineResource">${resourceIconHtml(def.resource,'sm')}</div>
      </div>

      ${atMax
        ? `<div class="combinedMineMax">Максимальный уровень станции достигнут</div>`
        : `
          <div class="combinedMineUpgrade">
            <div class="combinedMineNext">
              <span>Следующий уровень</span>
              <b>УР. ${nextLevel}</b>
              <small>Добыча станет +${nextRate.toFixed(2)} ${def.resource}/с</small>
            </div>

            <div class="combinedMineCostTitle">
              Стоимость улучшения
              <span>• каждый уровень +${Math.round(MINING_UPGRADE_COST_STEP * 100)}% от базовой цены</span>
            </div>

            <div class="costLine">
              ${Object.entries(upgradeCost).map(([r,a])=>resourceCostChipHtml(r,a)).join('')}
            </div>

            ${upgradeQueue ? buildProgressHtml(upgradeQueue,'Модернизация станции') : ''}

            <div class="combinedMineActions">
              ${upgradeQueue
                ? `<button class="cancelBuildBtn cancelMineUpgradeBtn" data-mine-upgrade="${type}">Отменить улучшение</button>`
                : `<button class="buildBtn startMineUpgradeBtn" data-mine-upgrade="${type}" ${canAfford(planet,upgradeCost)?'':'disabled'}>
                    Улучшить до Ур. ${nextLevel}
                  </button>`
              }
            </div>
          </div>
        `
      }
    </div>`;
  }

  return `<div class="buildCard ${type}">
    <div class="buildHead">
      ${buildingIconHtml(type)}
      <div class="buildText">
        <div class="buildName">${buildingLabel}</div>
        <div class="buildDesc">${def.desc}</div>
      </div>
    </div>

    <div class="costLine">
      ${Object.entries(def.cost).map(([r,a])=>resourceCostChipHtml(r,a)).join('')}
    </div>

    <div class="buildBottom">
      <div>
        <div class="buildTime">Строительство: ${def.time} сек.</div>
        ${productionInfo}
      </div>
      ${inQueue
        ? `<button class="cancelBuildBtn cancelBuildingBtn" data-building="${type}">Отменить</button>`
        : `<button class="buildBtn buildBuildingBtn" data-building="${type}" ${disabled?'disabled':''}>Построить</button>`
      }
    </div>

    ${inQueue ? buildProgressHtml(queue,'Осталось') : ''}
  </div>`;
}

function renderFleetTab(planet){
  const autoPatrolNotice = `
    <div class="planetNotice autoPatrolNotice">
      <b>Автопатруль:</b> новые военные корабли после постройки автоматически охраняют район своей планеты и остаются закреплены за её списком флота, даже когда физически кружат в патрульной зоне.

    </div>
  `;
  const localShips = fleetAtPlanet(planet).filter(s => s.owner === 'player');

  const localHtml = localShips.length
    ? localShips.map(ship => {
        let status = ship._aeRepairTargetId
          ? ('🔧 Ремонтирует «' + (scouts.find(x=>x.id===ship._aeRepairTargetId)?.name || 'корабль') + '»')
          : ship._aeRepairingBy
            ? ('🔧 Ремонт корпуса • ' + Math.ceil(ship.hp) + '/' + Math.ceil(ship.maxHp))
            : ship.researchMining
          ? (
              ship.researchMining.stage === 'mining'
                ? 'Добывает Ксенит • ' + Math.ceil(ship.researchMining.remaining) + ' сек.'
                : ship.researchMining.stage === 'returning'
                  ? (ship.researchMining.loopActive ? 'Возвращается с Ксенитом' : 'Отозван • возвращается на базу')
                  : 'Автоматический рейс на астероид'
            )
          : ship.scanState
            ? 'Сканирует • ' + Math.ceil(ship.scanState.remaining) + ' сек.' + (activePlanetScanners(ship.scanState.planet, ship.owner).length > 1 ? ' • ×' + activePlanetScanners(ship.scanState.planet, ship.owner).length : '')
            : ship.cooldownState
              ? 'Заряжает батареи • ' + Math.ceil(ship.cooldownState.remaining) + ' сек.'
              : ship.autoPatrolPlanet === planet && ship.patrolState
                ? (ship.combatTarget ? 'Автопатруль • перехватывает противника' : 'Автопатруль • охраняет район планеты')
                : 'На орбите • готов';
        return `<div class="fleetShip">
          <div class="fleetShipIcon ${ship.type === 'scout' ? 'satelliteSlot' : ''}" ${ship.type === 'scout' ? '' : `style="background-image:url(${shipArt(ship.type)})"`}>
            ${ship.type === 'scout' ? renderSatelliteIcon('size-list') : ''}
          </div>
          <div class="fleetShipText">
            <div class="fleetShipName">${renderTinyShipIcon(ship.type,'xs')}<span>${ship.name}</span></div>
            <div class="fleetShipStatus">${status}</div>
          </div>
        </div>`;
      }).join('')
    : '<div class="fleetEmpty">На орбите нет ваших кораблей.</div>';

  const categories = [
    ['Лёгкие',['scout','interceptor','corvette','frigate']],
    ['Тяжёлые',['destroyer','cruiser','battleship']],
    ['Защитные',['guardian']],
    ['Транспортные',['cargo']],
    ['Сервисные',['repair']],
    ['Стратегические',['researchMiner','colonizer']],
    ['Сверхтяжёлые',['dreadnought','titan']]
  ];

  planetMenuContent.innerHTML = `${autoPatrolNotice}
    <div class="planetMenuGroup theme-fleet-orbit">
      <div class="planetMenuGroupTitle">Флот на орбите</div>
      ${localHtml}
    </div>

      <div class="planetMenuGroup ${shipCategoryClass(title)}">

        ${types.map(type => shipBuildCardHtml(planet,type)).join('')}
      </div>
    `).join('')}

  planetMenuContent.querySelectorAll('.buildShipBtn').forEach(btn => {
    btn.addEventListener('click', e => {
      e.preventDefault();
      e.stopPropagation();
      startShipBuild(planet,btn.dataset.ship);
    });
  });
  planetMenuContent.querySelectorAll('.cancelShipBtn').forEach(btn => {
