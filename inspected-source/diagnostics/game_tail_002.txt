let cargoSourcePlanet = null;

function availableColonizersAtPlanet(planet){
  return fleetAtPlanet(planet).filter(ship =>
    ship.owner === 'player' &&
    ship.type === 'colonizer' &&
    !ship.resourceTransport
  );
}

function availableCargoShipsAtPlanet(planet){
  return fleetAtPlanet(planet).filter(ship =>
    ship.owner === 'player' &&
    ship.type === 'cargo' &&
    !ship.resourceTransport
  );
}

  return planets.filter(p => p.owner === 'player');
}

function refreshPlanetSpecialActions(planet, localFleet=fleetAtPlanet(planet)){
  const colonizers = localFleet.filter(s => s.owner === 'player' && s.type === 'colonizer');
  const usableColonizer = colonizers
    .filter(s => (s.colonists || 0) >= 100)
    .sort((a,b)=>(b.colonists||0)-(a.colonists||0))[0] || null;

  const canColonize =
    planet.owner === 'neutral' &&
    planet.playerScanned &&
    planet.habitable &&
    colonizers.length > 0;

  planetColonizeBtn.hidden = !canColonize;
  if(canColonize){
    const settlers = usableColonizer?.colonists || 0;
    planetColonizeBtn.disabled = !usableColonizer;
    planetColonizeBtn.textContent = usableColonizer
      ? 'Колонизировать • ' + settlers + ' колонистов'
      : 'На корабле недостаточно колонистов';
  }

  // Внутренняя логистика между собственными колониями теперь абстрагирована:
  // все добытые ресурсы входят в единый имперский резерв и доступны на любой колонии.
  // Грузовые корабли остаются в игре для будущих внешних/специальных логистических задач.
  planetCargoBtn.hidden = true;
  planetCargoBtn.disabled = true;
}

function colonizePlayerPlanet(planet){
  if(!planet || planet.owner !== 'neutral' || !planet.playerScanned || !planet.habitable) return;

  const ship = availableColonizersAtPlanet(planet)
    .filter(s => (s.colonists || 0) >= 100)
    .sort((a,b)=>(b.colonists||0)-(a.colonists||0))[0];

  if(!ship){
    pushEvent('На орбите нет готового колонизационного корабля с колонистами.', 'discovery');
    return;
  }

  const settlers = Math.max(100,Math.floor(ship.colonists || 0));

  planet.owner = 'player';
  planet.population = settlers;
  planet.civilizationLevel = 1;
  planet.playerScanned = true;
  planet.playerDiscoveredResources = [...planet.resources];
  planet.stockpile = colonyMineStarterStockpile(planet);
  planet.colonyStarterMineKit = true;

  // Колонизационный корабль становится первым колониальным центром.
  planet.buildings = {
    ironMine:0,copperMine:0,lithiumMine:0,
    foodFarm:1,waterPlant:1,habitat:1,clinic:0,entertainment:0,securityCenter:0,
    scienceCenter:0,
    missileBattery:0,laserBattery:0,shipyard:1
  };
  planet.upgrades = {
    foodFarm:0,waterPlant:0,habitat:0,clinic:0,entertainment:0,securityCenter:0,
    scienceCenter:0
  };
  planet.upgradeQueues = {};
  planet.buildingQueues = {};
  planet.buildingQueue = null;
  planet.shipQueue = null;
  planet.captureState = null;

  ship.consumedForColony = true;
  destroyShip(ship,null);

  // Владелец планеты изменился — один раз перестраиваем её визуальные метки/ауру.
  buildPlanets();
  buildMiniMapDots();
  PERF.cullingDirty = true;
  updateWorldCulling(true);

  closePlanetInfo();
  setResourceBarPlanet(planet);

  pushEvent(
    'КОЛОНИЯ ОСНОВАНА: «' + planet.name + '». Высажено ' + settlers +
    ' человек. Колонизационный корабль преобразован в базовый центр и верфь.',
    'success'
  );
  const starterMineCount = mineTypesForPlanet(planet).length;
  pushEvent(
    'КОЛОНИАЛЬНЫЙ КОМПЛЕКТ: на складе достаточно ресурсов, чтобы построить по одной добывающей станции для всех ' +
    starterMineCount + ' месторождений этой планеты.',
    'resource'
  );
  pushEvent(
    'ИМПЕРСКАЯ ЛОГИСТИКА: ресурсы этой колонии автоматически входят в общий резерв и доступны для строительства на любой вашей планете.',
    'resource'
  );

  currentPlanetTab = 'mining';
  openPlanetFleetMenu(planet);
}

function cargoInputRows(){
  return [...cargoResourceRows.querySelectorAll('.cargoResourceRow')];
}

function cargoSelectedAmounts(){
  const cargo = {};
  cargoInputRows().forEach(row => {
    const resource = row.dataset.resource;
    const input = row.querySelector('.cargoAmountInput');
    const amount = Math.max(0,Math.floor(Number(input?.value || 0)));
    if(amount > 0) cargo[resource] = amount;
  });
  return cargo;
}

function updateCargoLoadUi(){
  const ship = selectedCargoShip();
  const destination = selectedCargoDestination();
  const capacity = ship?.cargoCapacity || CARGO_DEFAULT_CAPACITY;
  const cargo = cargoSelectedAmounts();
  const total = Object.values(cargo).reduce((a,b)=>a+b,0);
  const pct = Math.min(100,total/capacity*100);
  const over = total > capacity;

  cargoCapacityText.textContent = total + ' / ' + capacity;
  cargoCapacityText.classList.toggle('over',over);
  cargoCapacityFill.style.width = pct + '%';
  cargoCapacityFill.classList.toggle('over',over);

  const resourceKinds = Object.keys(cargo).length;
  if(!destination){
    cargoRouteSummary.textContent = 'Выберите планету назначения.';
  }else if(total <= 0){
    cargoRouteSummary.textContent = 'Выберите количество ресурсов для отправки на «' + destination.name + '».';
  }else if(over){
    cargoRouteSummary.textContent = 'Трюм переполнен на ' + (total-capacity) + ' ед. Уменьшите груз.';
  }else{
    cargoRouteSummary.textContent =
      'Маршрут: «' + cargoSourcePlanet.name + '» → «' + destination.name +
      '» • ' + resourceKinds + ' вид. ресурсов • ' + total + ' ед.';
  }

  cargoDeliverySend.disabled = !ship || !destination || total <= 0 || over;
}

function renderCargoResourceRows(){
  if(!cargoSourcePlanet) return;

  const resources = Object.keys(cargoSourcePlanet.stockpile || {})
    .filter(r => Math.floor(getStock(cargoSourcePlanet,r)) > 0)
    .sort((a,b)=>getStock(cargoSourcePlanet,b)-getStock(cargoSourcePlanet,a));

  if(!resources.length){
    cargoResourceRows.innerHTML = '<div class="cargoEmpty">На этой планете сейчас нет ресурсов для отправки.</div>';
    updateCargoLoadUi();
    return;
  }

  cargoResourceRows.innerHTML = resources.map(resource => {
    const stock = Math.floor(getStock(cargoSourcePlanet,resource));
    return `
      <div class="cargoResourceRow" data-resource="${resource}">
        <div class="cargoResourceName">
          ${resourceIconHtml(resource,'sm')}
          <span>${resource}<small class="cargoResourceStock">На складе: ${stock}</small></span>
        </div>
        <input class="cargoAmountInput" type="number" min="0" max="${stock}" step="1" value="0" inputmode="numeric">
        <button class="cargoMaxBtn" type="button">МАКС</button>
      </div>
    `;
  }).join('');

  cargoInputRows().forEach(row => {
    const input = row.querySelector('.cargoAmountInput');
    const maxBtn = row.querySelector('.cargoMaxBtn');

    input.addEventListener('input',()=>{
      const stock = Math.floor(getStock(cargoSourcePlanet,row.dataset.resource));
      input.value = String(Math.max(0,Math.min(stock,Math.floor(Number(input.value || 0)))));
      updateCargoLoadUi();
    });

    maxBtn.addEventListener('click',()=>{
      const ship = selectedCargoShip();
      const capacity = ship?.cargoCapacity || CARGO_DEFAULT_CAPACITY;
      const currentCargo = cargoSelectedAmounts();
      const currentForThis = currentCargo[row.dataset.resource] || 0;
      const usedWithoutThis =
        Object.values(currentCargo).reduce((a,b)=>a+b,0) - currentForThis;
      const remaining = Math.max(0,capacity-usedWithoutThis);
      const stock = Math.floor(getStock(cargoSourcePlanet,row.dataset.resource));
      input.value = String(Math.min(stock,remaining));
      updateCargoLoadUi();
    });
  });

  updateCargoLoadUi();
}

function openCargoDelivery(sourcePlanet){
  if(!sourcePlanet || sourcePlanet.owner !== 'player') return;

  const cargoShips = availableCargoShipsAtPlanet(sourcePlanet);
  const destinations = playerColonies().filter(p => p !== sourcePlanet);

  if(!cargoShips.length){
    pushEvent('На орбите «' + sourcePlanet.name + '» нет свободного грузового корабля.', 'discovery');
    return;
  }
  if(!destinations.length){
    pushEvent('Для доставки ресурсов нужна хотя бы ещё одна ваша колония.', 'discovery');
    return;
  }

  cargoSourcePlanet = sourcePlanet;
  cargoSourceLabel.textContent = 'Отправление с «' + sourcePlanet.name + '»';

  cargoShipSelect.innerHTML = cargoShips.map(ship =>
    '<option value="' + ship.id + '">' + ship.name + ' • ' +
    (ship.cargoCapacity || CARGO_DEFAULT_CAPACITY) + ' ед.</option>'
  ).join('');

  cargoDestinationSelect.innerHTML = destinations.map(p =>
    '<option value="' + p.id + '">' + p.name + ' • население ' +
    Math.floor(p.population || 0).toLocaleString('ru-RU') + '</option>'
  ).join('');

  renderCargoResourceRows();
  cargoDeliveryOverlay.classList.add('open');
}

function startCargoDelivery(){
  if(!cargoSourcePlanet) return;

  const ship = selectedCargoShip();
  const destination = selectedCargoDestination();
  if(!ship || !destination || destination === cargoSourcePlanet) return;

  if(ship.orbitPlanet !== cargoSourcePlanet || ship.targetPlanet || ship.resourceTransport){
    pushEvent('Выбранный грузовой корабль уже занят или покинул орбиту.', 'discovery');
    closeCargoDelivery();
    return;
  }

  const capacity = ship.cargoCapacity || CARGO_DEFAULT_CAPACITY;
  const cargo = cargoSelectedAmounts();
  const total = Object.values(cargo).reduce((a,b)=>a+b,0);

  if(total <= 0 || total > capacity) return;

  for(const [resource,amount] of Object.entries(cargo)){
    if(amount > Math.floor(getStock(cargoSourcePlanet,resource))){
      pushEvent('На складе уже недостаточно ресурса «' + resource + '».', 'discovery');
      renderCargoResourceRows();
      return;
    }
  }

  for(const [resource,amount] of Object.entries(cargo)){
    addStock(cargoSourcePlanet,resource,-amount);
  }

  ship.resourceTransport = {
    sourcePlanet:cargoSourcePlanet,
    targetPlanet:destination,
    cargo:{...cargo},
    total,
    capacity,
    stage:'toTarget'
  };

  setScoutTarget(ship,destination,{silent:true,preserveTransport:true});
  updateSelectedPlanetResourceBar(true);

  pushEvent(
    ship.name + ' загрузил ' + total + ' ед. ресурсов и отправился: «' +
    cargoSourcePlanet.name + '» → «' + destination.name + '».',
    'resource'
  );

  scoutCommandHint.textContent =
    ship.name + ': доставка ' + total + ' ед. ресурсов на «' + destination.name + '».';
  scoutCommandHint.classList.add('show');
  clearTimeout(startCargoDelivery._timer);
  startCargoDelivery._timer = setTimeout(()=>scoutCommandHint.classList.remove('show'),3200);

  closeCargoDelivery();
  closePlanetInfo();
}

function finishResourceTransportArrival(ship,target){
  const mission = ship?.resourceTransport;
  if(!mission) return false;

  let unloadPlanet = null;
  let message = '';

  if(mission.stage === 'toTarget' && mission.targetPlanet === target){
    unloadPlanet = target;
    message =
      ship.name + ' доставил ' + mission.total + ' ед. ресурсов на «' + target.name + '».';
  }else if(mission.stage === 'returning' && mission.sourcePlanet === target){
    unloadPlanet = target;
    message =
      ship.name + ' вернулся на «' + target.name + '» и выгрузил перевозимый груз обратно на склад.';
  }else{
    return false;
  }

  for(const [resource,amount] of Object.entries(mission.cargo || {})){
    addStock(unloadPlanet,resource,amount);
  }

  ship.resourceTransport = null;
  pushEvent(message,'success');

  if(resourceBarPlanet === unloadPlanet){
    updateSelectedPlanetResourceBar(true);
  }

  if(currentFleetPlanet === unloadPlanet && planetFleetOverlay.classList.contains('open')){
    renderPlanetMenuTab(unloadPlanet);
  }

  scoutCommandHint.textContent = message;
  scoutCommandHint.classList.add('show');
  clearTimeout(finishResourceTransportArrival._timer);
  finishResourceTransportArrival._timer =
    setTimeout(()=>scoutCommandHint.classList.remove('show'),3200);

  return true;
}

function resumeResourceTransport(ship){
  const mission = ship?.resourceTransport;
  if(!mission || ship.destroyed) return false;

  const target = mission.stage === 'returning'
    ? mission.sourcePlanet
    : mission.targetPlanet;

  if(!target) return false;
  setScoutTarget(ship,target,{silent:true,preserveTransport:true});
  return true;
}

function fleetAtPlanet(planet){
  const ownColonies = planet?.owner === 'player' ? planets.filter(p=>p.owner==='player') : [];
  return scouts.filter(s => {
    if(!s || s.destroyed) return false;
    if(s.orbitPlanet === planet && !s.targetPlanet) return true;
    if(s.owner !== 'player' || planet.owner !== 'player') return false;

    // Патрульные корабли считаются флотом своей колонии даже когда физически
    // кружат вокруг неё или на короткое время перехватывают цель.
    if(s.autoPatrolPlanet === planet && !!s.patrolState) return true;
    if(s.homePlanet === planet && (!!s.patrolState || s.autoPatrolPlanet === planet)) return true;

    // Страховка для старых/переходных состояний: свободный корабль рядом с
    // собственной колонией относится к ближайшей колонии и не пропадает из списка.
    const busyElsewhere = !!s.targetPlanet || !!s.targetPoint || !!s.resourceTransport ||
      !!s.researchMining || !!s.planetMining || !!s._aeAnomalyTargetId || !!s._aeSignalDelivery;
    if(busyElsewhere || !ownColonies.length) return false;
    let nearest=null, best=Infinity;
    for(const p of ownColonies){
      const d=Math.hypot(s.x-p.x,s.y-p.y);
      if(d<best){best=d;nearest=p;}
    }
    return nearest===planet && best<=235;
  });
}

        // Если окно планеты появилось под уже зажатым пальцем, последующий click
        // не имеет собственного pointerdown на этой строке и будет проигнорирован.

          if(performance.now() < fleetSelectionUnlockAt){
            row._fleetSelectionArmed = false;
            e.preventDefault();
            e.stopPropagation();
            return;
          }
          row._fleetSelectionArmed = true;
          e.stopPropagation();
        }, {passive:false});

        row.addEventListener('pointercancel', () => {
          row._fleetSelectionArmed = false;
        });

        row.addEventListener('click', e => {
          if(
            performance.now() < fleetSelectionUnlockAt ||
            !row._fleetSelectionArmed
          ){
            row._fleetSelectionArmed = false;
            e.preventDefault();
            e.stopPropagation();
            return;
          }

          row._fleetSelectionArmed = false;

          if(e.target !== check) check.checked = !check.checked;

          if(check.checked) infoSelectedShipIds.add(ship.id);
          else infoSelectedShipIds.delete(ship.id);

          row.classList.toggle('selected', check.checked);
          updatePlanetTravelButton();
        });
      }
    });
  }

  refreshPlanetSpecialActions(planet,localFleet);

  if(planet.owner === 'player'){
    planetManageBtn.disabled = false;
    planetManageBtn.textContent = 'Управление планетой';
    if(gameSettings.actionHighlight) planetManageBtn.classList.add('manage-highlight');

  }

  planetInfoOverlay.classList.add('open');

  if(
    !firstColonizablePlanetTutorialShown &&
    gameSettings.hints &&
    planet.owner === 'neutral' &&

    planet.habitable

      'Постройте на одной из своих колоний <b>«Колонизационный корабль»</b>, посадите на него колонистов и отправьте к такой планете. Когда корабль окажется на орбите, в карточке появится кнопка <b>«Колонизировать»</b>. После основания колонии её добыча сразу подключится к <b>общему имперскому резерву</b>.'
    );
  }
}

  if(req.label === 'Население'){
    return Math.floor(req.current).toLocaleString('ru-RU') +

  }
  return req.current + (req.unit || '') + ' / ' + req.target + (req.unit || '');
}

function revealAlongScout(scout){
  if(scout.owner !== 'player') return;
  const d = Math.hypot(scout.x-scout.lastRevealX,scout.y-scout.lastRevealY);
  if(d >= 88){
    exploredAreas.push({x:scout.x,y:scout.y,r:150});
    scout.lastRevealX = scout.x;
    scout.lastRevealY = scout.y;
    drawFog();
  }
}


function shipDistance(a,b){
  return Math.hypot(a.x-b.x,a.y-b.y);
}

function findNearestEnemy(ship, radius){
  let best = null;
  let bestD = Infinity;
  for(const other of scouts){
    if(other === ship || other.destroyed || other.owner === ship.owner) continue;
    const d = shipDistance(ship,other);
    if(d <= radius && d < bestD){
      best = other;
      bestD = d;
    }
  }
  return best;
}

function patrolLeashRadius(patrol){
  if(!patrol) return 0;
  return patrol.chaseRadius || Math.max((patrol.radius || 175) + 110, (patrol.radius || 175) * 1.65);
}

function markPatrolReturn(ship,reason=''){
  const patrol = ship?.patrolState;
  if(!patrol) return false;
  ship.combatTarget = null;
  ship.attackAlerted = false;
  patrol.returning = true;
  patrol.chasing = false;
  patrol.chaseTargetId = null;
  patrol.detectionCooldown = Math.max(patrol.detectionCooldown || 0, 1.4);
  ensurePatrolVisual(ship);
  if(reason && ship.owner === 'player'){
    const now = gameElapsed || 0;
    if(!patrol.lastReturnNotice || now - patrol.lastReturnNotice > 4){
      patrol.lastReturnNotice = now;
      pushEvent(reason,'discovery');
    }
  }
  return true;
}

function beginCombat(attacker,target,fromPatrol=false){
  if(!attacker || !target || attacker.destroyed || target.destroyed) return;
  attacker.combatTarget = target;
  attacker.targetPlanet = null;
  attacker.orbitPlanet = null;
  attacker.researchMining = null;
  attacker.stationary = false;

  if(fromPatrol && attacker.patrolState){
    attacker.patrolState.returning = false;
    attacker.patrolState.chasing = true;
    attacker.patrolState.chaseTargetId = target.id;
  }

  if(fromPatrol && attacker.owner === 'player'){
    pushEvent(
      'ОБНАРУЖЕНИЕ: ' + attacker.name + ' заметил вражеский корабль «' + target.name + '» в зоне патруля.',
      'detection'
    );
  }
}

function combatWeaponStyle(ship){
  const type = ship?.type || '';
  if(['cruiser','battleship','dreadnought','titan'].includes(type)) return 'laser';
  if(['destroyer'].includes(type)) return 'rocket';
  if(['corvette','frigate','guardian'].includes(type)) return 'plasma';
  return 'pulse';
}
function combatHeavyShip(ship){
  return ['battleship','dreadnought','titan'].includes(ship?.type);
}
function fxOwnerClass(ship){
  return ship?.owner === 'enemy' ? 'enemy' : 'player';
}
function removeFxAfter(el,ms){
  if(!el) return;
  window.setTimeout(()=>el.remove(),ms);
}
function spawnCombatMuzzle(ship,angleDeg){
  const layer = combatFxLayer();
  if(!layer || combatFxMode()==='minimal') return;
  const el=document.createElement('div');
  el.className='combatFx combatMuzzle '+fxOwnerClass(ship);
  const rad=angleDeg*Math.PI/180;
  const offset=combatHeavyShip(ship)?12:8;
  el.style.left=(ship.x+Math.cos(rad)*offset)+'px';
  el.style.top=(ship.y+Math.sin(rad)*offset)+'px';
  layer.appendChild(el);
  removeFxAfter(el,220);
}
function spawnCombatImpact(target,sourceShip,heavy=false){
  const layer=combatFxLayer();
  if(!layer || !target || target.destroyed) return;
  const el=document.createElement('div');
  el.className='combatFx combatImpact '+fxOwnerClass(sourceShip)+(heavy?' heavy':'');
  el.style.left=target.x+'px';
  el.style.top=target.y+'px';
  layer.appendChild(el);
  removeFxAfter(el,380);
}
function spawnCombatLaser(attacker,target,heavy=false){
  const layer=combatFxLayer();
  if(!layer) return;
  const dx=target.x-attacker.x, dy=target.y-attacker.y;
  const dist=Math.hypot(dx,dy);
  const ang=Math.atan2(dy,dx)*180/Math.PI;
  const el=document.createElement('div');
  el.className='combatFx combatLaser '+fxOwnerClass(attacker)+(heavy?' heavy':'');
  el.style.left=attacker.x+'px';
  el.style.top=attacker.y+'px';
  el.style.width=Math.max(4,dist)+'px';
  el.style.transform='translateY(-50%) rotate('+ang+'deg)';
  layer.appendChild(el);
  spawnCombatMuzzle(attacker,ang);
  const impactDelay=combatFxMode()==='minimal'?45:90;
  window.setTimeout(()=>spawnCombatImpact(target,attacker,heavy),impactDelay);
  removeFxAfter(el,260);
}
function spawnCombatBolt(attacker,target,kind='pulse'){
  const layer=combatFxLayer();
  if(!layer) return;
  const sx=attacker.x, sy=attacker.y, tx=target.x, ty=target.y;
  const dx=tx-sx,dy=ty-sy,dist=Math.hypot(dx,dy);
  const ang=Math.atan2(dy,dx)*180/Math.PI;
  const el=document.createElement('div');
  el.className='combatFx combatBolt '+fxOwnerClass(attacker)+' '+kind;
  el.style.left=sx+'px';
  el.style.top=sy+'px';
  el.style.transform='translate(-50%,-50%) rotate('+ang+'deg)';
  layer.appendChild(el);
  spawnCombatMuzzle(attacker,ang);
  const mode=combatFxMode();
  let duration=kind==='rocket'?210:(kind==='plasma'?155:115);
  if(mode==='minimal') duration=Math.max(70,duration*.62);
  const anim=el.animate(
    [
      {left:sx+'px',top:sy+'px',opacity:1},
      {left:tx+'px',top:ty+'px',opacity:.95}
    ],
    {duration,easing:kind==='rocket'?'cubic-bezier(.18,.68,.28,1)':'linear',fill:'forwards'}
  );
  const done=()=>{
    spawnCombatImpact(target,attacker,kind==='rocket');
    el.remove();
  };
  if(anim?.finished) anim.finished.then(done).catch(()=>el.remove());
  else window.setTimeout(done,duration);
}
function spawnCombatShot(attacker,target){
  if(!attacker || !target || attacker.destroyed || target.destroyed) return;
  // Бои за пределами экрана продолжают считаться, но не создают невидимые DOM-анимации.
  if(typeof isWorldPointNearViewport==='function' &&
     !isWorldPointNearViewport(attacker.x,attacker.y,180) &&
     !isWorldPointNearViewport(target.x,target.y,180)) return;
  const layer=combatFxLayer();
  const mode=combatFxMode();
  const cap=mode==='max'?150:(mode==='minimal'?45:95);
  if(layer && layer.childElementCount>=cap) return;
  const style=combatWeaponStyle(attacker);
  if(style==='laser') spawnCombatLaser(attacker,target,combatHeavyShip(attacker));
  else if(style==='rocket') spawnCombatBolt(attacker,target,'rocket');
  else if(style==='plasma') spawnCombatBolt(attacker,target,'plasma');
  else spawnCombatBolt(attacker,target,'pulse');
}
function spawnCombatExplosion(ship){
  const layer=combatFxLayer();
  if(!layer || !ship) return;
  if(typeof isWorldPointNearViewport==='function' && !isWorldPointNearViewport(ship.x,ship.y,220)) return;
  const heavy=combatHeavyShip(ship);
  const mode=combatFxMode();

  const core=document.createElement('div');
  core.className='combatFx combatExplosion'+(heavy?' heavy':'');
  core.style.left=ship.x+'px';
  core.style.top=ship.y+'px';
  layer.appendChild(core);
  removeFxAfter(core,760);

  if(mode!=='minimal'){
    const ring=document.createElement('div');
    ring.className='combatFx combatExplosionRing'+(heavy?' heavy':'');
    ring.style.left=ship.x+'px';
    ring.style.top=ship.y+'px';
    layer.appendChild(ring);
    removeFxAfter(ring,820);
  }

  if(mode!=='minimal'){
    const count=mode==='max'?(heavy?13:9):(heavy?8:5);
    for(let i=0;i<count;i++){
      const d=document.createElement('div');
      d.className='combatFx combatDebris';
      d.style.left=ship.x+'px';
      d.style.top=ship.y+'px';
      const a=Math.random()*Math.PI*2;
      const r=(heavy?28:19)*(0.55+Math.random()*.9);
      d.style.setProperty('--dx',(Math.cos(a)*r).toFixed(1)+'px');
      d.style.setProperty('--dy',(Math.sin(a)*r).toFixed(1)+'px');
      layer.appendChild(d);
      removeFxAfter(d,820);
    }
  }
}

function damageShip(target,amount,attacker){
  if(!target || target.destroyed) return;
  if(attacker?.owner === 'enemy') amount *= getAIProfile().combat;
  target.hp = Math.max(0,(target.hp ?? target.maxHp ?? 25)-amount);

  if(target.owner === 'player'){
    if(!target.attackAlerted){
      target.attackAlerted = true;
      pushEvent(
        'ТРЕВОГА: «' + target.name + '» атакован кораблём «' + attacker.name + '».',
        'attack'
      );
    }
  }

  if(target.hp <= 0){
    destroyShip(target,attacker);
  }
}

function destroyShip(ship,killer){
  if(!ship || ship.destroyed) return;

  if(ship.owner === 'player' && ship.resourceTransport?.total > 0){
    pushEvent(
      'ПОТЕРЯ ГРУЗА: вместе с «' + ship.name + '» потеряно ' +
      ship.resourceTransport.total + ' ед. перевозимых ресурсов.',
      'attack'
    );
  }

  spawnCombatExplosion(ship);
  ship.destroyed = true;
  removeShipFromFleetGroups(ship.id);

  if(ship.targetPlanet?.id && ship.owner === 'enemy'){
    enemyReservedPlanetIds.delete(ship.targetPlanet.id);
  }

  removePatrolVisual(ship);
  ship.el?.remove();
  ship.timerEl?.remove();

  const index = scouts.indexOf(ship);
  if(index >= 0) scouts.splice(index,1);

  scouts.forEach(s => {
    if(s.combatTarget === ship){
      s.combatTarget = null;
      s.attackAlerted = false;
      if(s.patrolState){
        s.patrolState.returning = true;
        s.patrolState.chasing = false;
        s.patrolState.chaseTargetId = null;
        ensurePatrolVisual(s);
      }
    }
  });

  if(ship.owner === 'player' && !ship.consumedForColony){
    pushEvent('ПОТЕРЯ: корабль «' + ship.name + '» уничтожен.', 'attack');
  }else if(killer?.owner === 'player'){
    pushEvent('Вражеский корабль «' + ship.name + '» уничтожен.', 'success');
  }
}

function updateCombat(ship,dtGame){
  const target = ship.combatTarget;
  if(!target || target.destroyed){
    ship.combatTarget = null;
    ship.attackAlerted = false;
    if(ship.patrolState){
      markPatrolReturn(ship);
    }else if(ship.resourceTransport){
      resumeResourceTransport(ship);
    }
    return;
  }

  // Корабль, поднятый по тревоге с патруля, не должен преследовать цель через всю карту.
  // Если цель покинула допустимую область преследования, корабль разрывает контакт и
  // физически возвращается в свою патрульную зону.
  if(ship.patrolState){
    const patrol = ship.patrolState;
    const leash = patrolLeashRadius(patrol);
    const targetFromCenter = Math.hypot(target.x-patrol.cx,target.y-patrol.cy);
    const shipFromCenter = Math.hypot(ship.x-patrol.cx,ship.y-patrol.cy);
    if(targetFromCenter > leash || shipFromCenter > leash + 45){
      const targetName = target.name || 'цель';
      markPatrolReturn(
        ship,
        'ПАТРУЛЬ: «' + targetName + '» ушёл слишком далеко. «' + ship.name + '» прекращает погоню и возвращается в свою зону.'
      );
      return;
    }
  }

  // Разведчики, грузовые, колонизационные, добывающие и ремонтные суда не являются
  // боевыми единицами: они могут быть атакованы, но не ведут встречный огонь.
  if(!isMilitaryShipType(ship.type)){
    ship.combatTarget = null;
    ship.attackAlerted = false;
    if(ship.patrolState) markPatrolReturn(ship);
    return;
  }

  const dx = target.x-ship.x;
  const dy = target.y-ship.y;
  const dist = Math.hypot(dx,dy);

  // Даже если корабль уже находится в радиусе стрельбы и больше не двигается,
  // он должен смотреть носом на текущую боевую цель.
  if(dist > .001){
    ship._headingDeg = Math.atan2(dy,dx) * 180 / Math.PI;
  }

  if(dist > ship.attackRange){
    const step = shipMoveSpeed(ship) * 1.15 * dtGame;
    if(dist > .001){
      ship.x += dx/dist*Math.min(step,dist);
      ship.y += dy/dist*Math.min(step,dist);
    }
    return;
  }

  ship.attackCooldown = Math.max(0,(ship.attackCooldown || 0)-dtGame);
  if(ship.attackCooldown <= 0){
    ship.attackCooldown = ship.attackInterval || 1.5;
    const shotDamage = Number.isFinite(ship.damage) ? ship.damage : 2;
    if(shotDamage > 0){
      spawnCombatShot(ship,target);
      damageShip(target,shotDamage,ship);
    }

    // Только настоящий военный корабль отвечает огнём. Поэтому разведчик или транспорт,
    // уничтоженный перехватчиком, больше не успевает "забрать его с собой" ответным залпом.
    if(!target.destroyed && !target.combatTarget && isMilitaryShipType(target.type)){
      target.combatTarget = ship;
      target.targetPlanet = null;
      target.orbitPlanet = null;
      target.researchMining = null;
      target.stationary = false;
    }
  }
}

function updatePatrol(ship,dtGame){
  const patrol = ship.patrolState;
  if(!patrol) return;

  if(ship.combatTarget){
    updateCombat(ship,dtGame);
    return;
  }

  // После погони корабль не телепортируется обратно на орбиту патруля, а реально летит назад.
  if(patrol.returning){
    const relX = ship.x-patrol.cx;
    const relY = (ship.y-patrol.cy)/.70;
    const rejoinAngle = Math.abs(relX)+Math.abs(relY) > .001 ? Math.atan2(relY,relX) : (patrol.angle || 0);
    const tx = patrol.cx + Math.cos(rejoinAngle)*patrol.orbitRadius;
    const ty = patrol.cy + Math.sin(rejoinAngle)*patrol.orbitRadius*.70;
    const dx = tx-ship.x, dy = ty-ship.y;
    const dist = Math.hypot(dx,dy);
    const step = shipMoveSpeed(ship) * dtGame;
    if(dist <= Math.max(step,3)){
      ship.x = tx; ship.y = ty;
      patrol.angle = rejoinAngle;
      patrol.returning = false;
      patrol.chasing = false;
      patrol.chaseTargetId = null;
      patrol.detectionCooldown = Math.max(patrol.detectionCooldown || 0,1.0);
      ensurePatrolVisual(ship);
    }else if(dist > .001){
      ship._headingDeg = Math.atan2(dy,dx) * 180 / Math.PI;
      ship.x += dx/dist*Math.min(step,dist);
      ship.y += dy/dist*Math.min(step,dist);
    }
    return;
  }

  patrol.angle += .22 * patrol.direction * dtGame;
  ship.x = patrol.cx + Math.cos(patrol.angle)*patrol.orbitRadius;
  ship.y = patrol.cy + Math.sin(patrol.angle)*patrol.orbitRadius*.70;

  patrol.detectionCooldown = Math.max(0,(patrol.detectionCooldown || 0)-dtGame);
  if(patrol.detectionCooldown <= 0){
    patrol.detectionCooldown = 1.1;
    const enemy = findNearestEnemy(ship,patrol.radius);
    if(enemy){
      removePatrolVisual(ship);
      beginCombat(ship,enemy,true);
    }
  }
}

function updateScouts(dt){
  if(!gameStarted) return;
  const stepScale = dt * gameSpeed;

  // Копия нужна, потому что корабль может быть уничтожен во время цикла.
  for(const scout of [...scouts]){
    if(scout.destroyed) continue;

    // Запоминаем фактическую позицию в начале кадра. В конце кадра по смещению
    // определяем направление движения и поворачиваем нос корабля точно по курсу.
    const previousX = scout.x;
    const previousY = scout.y;

    if(scout.cooldownState){
      updateScoutCooldown(scout,stepScale);
    }

    if(scout.combatTarget){
      updateCombat(scout,stepScale);
    }else if(scout.patrolState){
      updatePatrol(scout,stepScale);
    }else if(scout.targetPoint){
      const target = scout.targetPoint;
      const dx = target.x - scout.x;
      const dy = target.y - scout.y;
      const dist = Math.hypot(dx,dy);
      const step = shipMoveSpeed(scout) * stepScale;
      if(dist <= Math.max(step,3)){
        scout.x = target.x;
        scout.y = target.y;
        scout.targetPoint = null;
        scout.stationary = true;
        clearWaypointMarker(scout);
        if(scout.owner === 'player') pushEvent(scout.name + ' прибыл в указанную точку и ожидает приказа.', 'discovery');
      }else{
        scout.x += dx / dist * step;
        scout.y += dy / dist * step;
      }
    }else if(scout.targetPlanet){
      const target = scout.targetPlanet;
      const dx = target.x - scout.x;
      const dy = target.y - scout.y;
      const dist = Math.hypot(dx,dy);
      const step = shipMoveSpeed(scout) * stepScale;

      if(dist <= Math.max(step, 3)){
        if(target.kind === 'researchAsteroid'){
          beginOrbit(scout,target);
          if(scout.owner === 'player' && scout.canMineResearch){
            if(getAsteroidMinerCount(target, scout) > 2){
              scoutCommandHint.textContent = 'На астероиде уже работают 2 добытчика.';
              scoutCommandHint.classList.add('show');
              setTimeout(() => scoutCommandHint.classList.remove('show'), 2400);
            }else if(scout.researchMining && scout.researchMining.asteroid && scout.researchMining.asteroid.id === target.id){
              scout.researchMining.stage = 'mining';
              scout.researchMining.remaining = scout.researchMining.miningTime || 14;
              scoutCommandHint.textContent = scout.name + ' начал новый цикл добычи на астероиде.';
              scoutCommandHint.classList.add('show');
              setTimeout(() => scoutCommandHint.classList.remove('show'), 2400);
            }else{
              startResearchMining(scout,target);
              scoutCommandHint.textContent = scout.name + ' начал добычу Ксенита.';
              scoutCommandHint.classList.add('show');
              setTimeout(() => scoutCommandHint.classList.remove('show'), 2600);
            }
          }
        }else if(scout.owner === 'player'){
          if(scout.canScan){
            exploredAreas.push({x:target.x, y:target.y, r:230});
            drawFog();
          }

          beginOrbit(scout, target);

          const isResourceDelivery = finishResourceTransportArrival(scout,target);

          const isMiningReturn = !!(
            scout.canMineResearch &&
            scout.researchMining &&
            scout.researchMining.stage === 'returning' &&
            scout.researchMining.homePlanet &&
            scout.researchMining.homePlanet.id === target.id
          );
          const isPlanetMiningReturn = !!(
            scout.planetMining &&
            scout.planetMining.stage === 'returning' &&
            scout.planetMining.homePlanet &&
            scout.planetMining.homePlanet.id === target.id
          );
          const isPlanetMiningTarget = !!(
            scout.canMinePlanet &&
            target.owner === 'neutral' &&
            !target.habitable &&
            target.playerScanned
          );

          if(isResourceDelivery){
            // Груз автоматически выгружен. Корабль остаётся на орбите и готов к новому рейсу.
          }else if(isMiningReturn){
            depositAsteroidCargo(scout, target);

            scoutCommandHint.textContent = scout.researchMining
              ? scout.name + ' выгрузил ресурс и снова отправился к астероиду.'
              : scout.name + ' вернулся на базу и завершил автоматические рейсы.';
            scoutCommandHint.classList.add('show');
            clearTimeout(updateScouts._timer);
            updateScouts._timer = setTimeout(() => scoutCommandHint.classList.remove('show'), 3200);
          }else if(isPlanetMiningReturn){
            depositPlanetMiningCargo(scout,target);
            scoutCommandHint.textContent = scout.planetMining
              ? scout.name + ' выгрузил ресурсы и снова отправился на добычу.'
              : scout.name + ' вернулся на базу и завершил добывающие рейсы.';
            scoutCommandHint.classList.add('show');
            clearTimeout(updateScouts._timer);
            updateScouts._timer = setTimeout(() => scoutCommandHint.classList.remove('show'), 3200);
          }else if(isPlanetMiningTarget){
            if(getPlanetMinerCount(target,scout) >= 2 && !(scout.planetMining && scout.planetMining.planet?.id === target.id)){
              scoutCommandHint.textContent = 'На этой планете уже работают 2 грузовых корабля.';
              scoutCommandHint.classList.add('show');
              setTimeout(()=>scoutCommandHint.classList.remove('show'),2400);
            }else if(scout.planetMining && scout.planetMining.planet?.id === target.id){
              scout.planetMining.stage = 'mining';
              scout.planetMining.remaining = scout.planetMining.miningTime || 18;
              scoutCommandHint.textContent = scout.name + ' начал новый цикл загрузки ресурсов.';
              scoutCommandHint.classList.add('show');
              setTimeout(()=>scoutCommandHint.classList.remove('show'),2400);
            }else if(startPlanetResourceMining(scout,target)){
              scoutCommandHint.textContent = scout.name + ' начал загрузку ресурсов для доставки в империю.';
              scoutCommandHint.classList.add('show');
              setTimeout(()=>scoutCommandHint.classList.remove('show'),2600);
            }
          }else{
            startPlanetScan(scout, target);

          if(scout.scanState){
            scoutCommandHint.textContent =
              scout.name + ' вышел на орбиту. Начато изучение планеты: 60 сек.';
            pushEvent(
              scout.name + ' достиг планеты и вышел на орбиту. Запущено сканирование.',
              'discovery'
            );
          }else{
            scoutCommandHint.textContent =
              scout.name + ' достиг планеты и вышел на орбиту.';
            pushEvent(
              scout.name + ' прибыл к «' + target.name + '».',
              'discovery'
            );
          }

          scoutCommandHint.classList.add('show');
          clearTimeout(updateScouts._timer);
          updateScouts._timer = setTimeout(() => scoutCommandHint.classList.remove('show'), 3500);
          }
        }else{
          const arrivedId = target.id;
          enemyReservedPlanetIds.delete(arrivedId);
          if(
            scout.type === 'colonizer' &&
            target.owner === 'neutral' &&
            target.habitable &&
            enemyVisitedPlanetIds.has(target.id)
          ){
            enemyColonizePlanet(scout,target);
            continue;
          }
          beginOrbit(scout, target);
          startPlanetScan(scout, target);
        }
      }else{
        scout.x += dx / dist * step;
        scout.y += dy / dist * step;
        revealAlongScout(scout);
      }
    }else if(scout.orbitPlanet){
      scout.orbitAngle += scout.orbitAngularSpeed * scout.orbitDirection * stepScale;
      placeScoutOnOrbit(scout);

      if(scout.researchMining){
        updateResearchMining(scout, stepScale);
      }else if(scout.planetMining){
        updatePlanetResourceMining(scout, stepScale);
      }else if(scout.scanState){
        updatePlanetScan(scout, stepScale);
      }else if(
        scout.owner === 'enemy' &&
        scout.type === 'scout' &&
        !scout.cooldownState
      ){
        scout.idleTimer -= stepScale;
        if(scout.idleTimer <= 0){
          chooseEnemyScoutTarget(scout);
          scout.idleTimer = enemyScoutDecisionDelay();
        }
      }
    }else if(scout.stationary){
      // Ничего не делаем — корабль стоит в точке до нового приказа.
    }

    const movedX = scout.x - previousX;
    const movedY = scout.y - previousY;
    if(Math.hypot(movedX,movedY) > 0.015){
      scout._headingDeg = Math.atan2(movedY,movedX) * 180 / Math.PI;
    }

    revealAlongScout(scout);
    updateScoutElement(scout);
  }
}


function aiStartBuilding(planet,type){
  const def = BUILDING_DEFS[type];
  if(!def) return false;
  if(!planet.buildingQueues) planet.buildingQueues = {};
  if(planet.buildingQueues[type]) return false;
  if(def.single && (planet.buildings[type] || 0) >= 1) return false;
  if(def.resource && !planet.resources.includes(def.resource)) return false;
  if(!spendCost(planet,def.cost)) return false;
  planet.buildingQueues[type] = {type,remaining:def.time,total:def.time};
  return true;
}
function aiQueueShip(planet,type){
  const def = SHIP_DEFS[type];
  if(!def || planet.shipQueue || !spendCost(planet,def.cost)) return false;
  planet.shipQueue = {type,remaining:def.time,total:def.time,pending:[]};
  return true;
}
function enableEnemyPatrolQuiet(ship,aroundPlanet=null){
  if(!ship || ship.destroyed) return;
  const cx = aroundPlanet?.x ?? ship.x;
  const cy = aroundPlanet?.y ?? ship.y;
  ship.targetPlanet = null;
  ship.targetPoint = null;
  ship.orbitPlanet = null;
  ship.researchMining = null;
  ship.combatTarget = null;
  ship.stationary = false;
  const patrolRadius = gameSettings.aiDifficulty === 'aggressive' ? 235 : 195;
  ship.patrolState = {
    cx,cy,
    radius:patrolRadius,
    chaseRadius:Math.max(patrolRadius + 120, patrolRadius * 1.65),
    orbitRadius:58,
    angle:Math.random()*Math.PI*2,
    direction:Math.random()<.5?-1:1,
    detectionCooldown:0,
    returning:false,
    chasing:false,
    chaseTargetId:null
  };
  if(!gameSettings.fog) ensurePatrolVisual(ship);
}
function findEnemyColonizationTarget(){
  return planets
    .filter(p=>p.owner === 'neutral' && p.habitable && enemyVisitedPlanetIds.has(p.id))
    .sort((a,b)=>Math.hypot(a.x-enemyPlanet.x,a.y-enemyPlanet.y)-Math.hypot(b.x-enemyPlanet.x,b.y-enemyPlanet.y))[0] || null;
}
function enemyCombatShips(){
  return scouts.filter(s=>s.owner === 'enemy' && !['scout','cargo','colonizer','researchMiner'].includes(s.type));
}
function assignEnemyShipOrder(ship){
  if(!ship || ship.destroyed) return;
  if(ship.type === 'scout'){
    ship.idleTimer = Math.max(1.5,enemyScoutDecisionDelay()*.35);
    return;
  }
  if(ship.type === 'colonizer'){
    const target = findEnemyColonizationTarget();
    if(target){ setScoutTarget(ship,target,{silent:true}); }
    return;
  }
  if(['cargo','researchMiner'].includes(ship.type)) return;

  // Свежепостроенный военный корабль больше НИКОГДА не летит к игроку сам по себе.
  // Сначала он остаётся в обороне своей колонии. Наступление запускается отдельно,
  // только после фазы развития и только сформированной группой кораблей.
  enableEnemyPatrolQuiet(ship,ship.orbitPlanet || ship.homePlanet || enemyPlanet);
}

function enemyEconomyDevelopmentScore(){
  const colonies = planets.filter(p=>p.owner === 'enemy');
  let mineLevels = 0;
  let infrastructure = 0;
  colonies.forEach(p=>{
    mineLevels += (p.buildings?.ironMine||0) + (p.buildings?.copperMine||0) + (p.buildings?.lithiumMine||0);
    infrastructure += (p.buildings?.foodFarm||0) + (p.buildings?.waterPlant||0) + (p.buildings?.habitat||0) + (p.buildings?.scienceCenter||0);
  });
  return {colonies,mineLevels,infrastructure};
}

function enemyAttackCandidates(){
  return enemyCombatShips().filter(ship=>{
    if(!ship || ship.destroyed || ship.combatTarget || ship.targetPlanet || ship.targetPoint) return false;
    // Уже находящиеся у территории игрока корабли не забираем в новую волну.
    const patrolNearPlayer = ship.patrolState && Math.hypot(ship.patrolState.cx-playerPlanet.x,ship.patrolState.cy-playerPlanet.y) < 500;
    return !patrolNearPlayer;
  });
}

function launchEnemyAttackWave(){
  const profile = getAIProfile();

  // В начале партии ИИ обязан развиваться, а не рашить стартовую планету игрока.
  if(gameElapsed < profile.peaceTime) return false;
  if(enemyAIState.attackCooldown > 0) return false;

  const development = enemyEconomyDevelopmentScore();
  // Нормальная атака начинается после появления второй колонии или заметного развития добычи.
  // Есть страховка от вечного бездействия: спустя ещё 4 минуты одной развитой стартовой базы достаточно.
  const economyReady = development.colonies.length >= 2 || development.mineLevels >= 5;
  if(!economyReady && gameElapsed < profile.peaceTime + 240) return false;

  const candidates = enemyAttackCandidates();
  // ИИ всегда оставляет хотя бы одного защитника на каждую свою колонию.
  const defendersToKeep = Math.max(1,development.colonies.length);
  const availableForAttack = Math.max(0,candidates.length - defendersToKeep);
  if(availableForAttack < profile.attackFleetMin) return false;

  const militaryBias = gameSettings.aiPersonality === 'militarist' ? 1.35 : gameSettings.aiPersonality === 'economist' ? .55 : 1;
  const launchChance = Math.min(.72,(.16 + profile.aggression*.72) * militaryBias);
  if(Math.random() >= launchChance) return false;

  const waveSize = Math.min(profile.attackWaveMax,availableForAttack);
  // Более быстрые корабли не должны убегать вперёд по одному: выбираем одну общую ударную группу.
  const wave = candidates
    .sort((a,b)=>Math.hypot(a.x-enemyPlanet.x,a.y-enemyPlanet.y)-Math.hypot(b.x-enemyPlanet.x,b.y-enemyPlanet.y))
    .slice(0,waveSize);

  const angle = Math.random()*Math.PI*2;
  const distance = 145 + Math.random()*70;
  const target = {
    x:playerPlanet.x + Math.cos(angle)*distance,
    y:playerPlanet.y + Math.sin(angle)*distance
  };

  wave.forEach(ship=>{
    ship.targetPlanet = null;
    ship.targetPoint = {...target};
    ship.orbitPlanet = null;
    ship.researchMining = null;
    ship.combatTarget = null;
    ship.stationary = false;
    ship.patrolState = null;
    ship.autoPatrolPlanet = null;
    removePatrolVisual(ship);
  });

  enemyAIState.attackCooldown = profile.attackCooldown;
  if(!gameSettings.fog){
    pushEvent('Наблюдение: противник сформировал ударную группу из ' + wave.length + ' кораблей.','discovery');
  }
  return true;
}
function enemyColonizePlanet(ship,planet){
  if(!ship || !planet || planet.owner !== 'neutral') return false;
  planet.owner = 'enemy';
  planet.population = Math.max(100,Math.floor(ship.colonists || 300));
  planet.stockpile = {'Железо':260,'Медь':150,'Литий':100,'Алюминий':60};
  planet.buildings.shipyard = 1;
  planet.buildings.foodFarm = 1;
  planet.buildings.waterPlant = 1;
  planet.buildings.habitat = 1;
  updatePlanetMapData(planet);
  buildMiniMapDots();
  if(!gameSettings.fog){
    pushEvent('Наблюдение: ИИ основал колонию «' + planet.name + '».','discovery');
  }
  destroyShip(ship,null);
  return true;
}
function enemyAIChooseShip(planet){
  const profile = getAIProfile();
  const enemyPlanets = planets.filter(p=>p.owner === 'enemy');
  const queuedScouts = enemyPlanets.filter(p=>p.shipQueue?.type === 'scout').length;
  const queuedColonizers = enemyPlanets.filter(p=>p.shipQueue?.type === 'colonizer').length;
  const queuedCombat = enemyPlanets.filter(p=>p.shipQueue && !['scout','cargo','colonizer','researchMiner'].includes(p.shipQueue.type)).length;
  const ownScouts = scouts.filter(s=>s.owner === 'enemy' && s.type === 'scout').length + queuedScouts;
  const ownColonizers = scouts.filter(s=>s.owner === 'enemy' && s.type === 'colonizer').length + queuedColonizers;
  const enemyColonies = enemyPlanets.length;
  const ownCombat = enemyCombatShips().length + queuedCombat;
  const colonyTarget = findEnemyColonizationTarget();

  // Первые решения ИИ — разведка и расширение, а не спам боевых кораблей.
  if(ownScouts < (gameSettings.aiDifficulty === 'easy' ? 2 : 3)) return 'scout';
  if(colonyTarget && ownColonizers === 0 && enemyColonies < profile.maxColonies){
    if(gameSettings.aiPersonality !== 'militarist' || Math.random() < .70) return 'colonizer';
  }

  // Пока действует стартовая фаза развития, ограничиваем военный флот.
  // Стартовый перехватчик входит в этот лимит.
  if(gameElapsed < profile.peaceTime && ownCombat >= profile.earlyCombatCap) return null;

  const pools = {
    economist:['interceptor','corvette','cargo','guardian','frigate'],
    balanced:['interceptor','corvette','frigate','guardian','destroyer'],
    militarist:['interceptor','corvette','frigate','destroyer','cruiser']
  };
  const pool = pools[gameSettings.aiPersonality] || pools.balanced;
  for(let i=pool.length-1;i>=0;i--){
    const type = pool[Math.floor(Math.random()*(i+1))];
    if(canAfford(planet,SHIP_DEFS[type].cost)) return type;
  }
  return canAfford(planet,SHIP_DEFS.interceptor.cost) ? 'interceptor' : null;
}
function enemyAIThink(){
  const profile = getAIProfile();
  const colonies = planets.filter(p=>p.owner === 'enemy');

  colonies.forEach(planet=>{
    if(!planet.buildingQueues) planet.buildingQueues = {};

    if(!Object.keys(planet.buildingQueues).length){
      const mineTargets = [
        ['Железо','ironMine'],['Медь','copperMine'],['Литий','lithiumMine']
      ].filter(([resource,type])=>planet.resources.includes(resource) && (planet.buildings[type]||0) < (gameSettings.aiPersonality === 'economist' ? 3 : 2));

      let built = false;
      for(const [,type] of mineTargets){
        if(aiStartBuilding(planet,type)){ built = true; break; }
      }
      if(!built && gameSettings.aiPersonality === 'economist'){
        for(const type of ['foodFarm','waterPlant','habitat','scienceCenter']){
          if((planet.buildings[type]||0) < (type === 'scienceCenter' ? 1 : 2) && aiStartBuilding(planet,type)) break;
        }
      }
    }

    if(!planet.shipQueue){
      const type = enemyAIChooseShip(planet);
      if(type) aiQueueShip(planet,type);
    }
  });

  // Корабли, долетевшие в боевую точку, переходят в патруль и ищут цели.
  enemyCombatShips().forEach(ship=>{
    if(ship.stationary && !ship.patrolState && !ship.combatTarget){
      enableEnemyPatrolQuiet(ship,null);
    }
  });

  // Наступление — отдельное стратегическое решение, а не приказ каждому новому кораблю.
  launchEnemyAttackWave();

  enemyAIState.thinkTimer = profile.think * (.86 + Math.random()*.28);
}
function updateEnemyAI(dtGame){
  if(!gameStarted) return;
  const profile = getAIProfile();
  const colonies = planets.filter(p=>p.owner === 'enemy');
  enemyAIState.attackCooldown = Math.max(0,(enemyAIState.attackCooldown||0) - dtGame);

  colonies.forEach(p=>{
    // Базовая экономика ИИ. Сложность определяет ресурсный темп.
    addStock(p,'Железо',.34*profile.income*dtGame);
    addStock(p,'Медь',.22*profile.income*dtGame);
    addStock(p,'Литий',.14*profile.income*dtGame);
    addStock(p,'Алюминий',.055*profile.income*dtGame);

    for(const [type,def] of Object.entries(BUILDING_DEFS)){
      if(def.resource && def.production){
        const count = p.buildings?.[type] || 0;
        if(count) addStock(p,def.resource,count*def.production*profile.income*dtGame);
      }
    }

    for(const [type,q] of Object.entries({...p.buildingQueues || {}})){
      q.remaining -= dtGame*profile.buildSpeed;
      if(q.remaining <= 0){
        delete p.buildingQueues[type];
        p.buildings[type] = (p.buildings[type]||0)+1;
        if(!gameSettings.fog) pushEvent('Наблюдение: ИИ завершил «'+BUILDING_DEFS[type].name+'» на «'+p.name+'».','discovery');
      }
    }

    if(p.shipQueue){
      p.shipQueue.remaining -= dtGame*profile.buildSpeed;
      if(p.shipQueue.remaining <= 0){
        const type = p.shipQueue.type;
        p.shipQueue = null;
        const ship = makeBuiltShip('enemy',p,type);
        if(!gameSettings.fog) pushEvent('Наблюдение: ИИ построил корабль «'+ship.name+'».','discovery');
        assignEnemyShipOrder(ship);
      }
    }
  });

  enemyAIState.thinkTimer -= dtGame;
  if(enemyAIState.thinkTimer <= 0) enemyAIThink();
}

function gameLoop(now){
  const effectiveFps = gameSettings.powerSaver ? 30 : Number(gameSettings.fps || 60);
  const minFrameMs = effectiveFps <= 30 ? 1000/30 : 0;

  if(minFrameMs && now-lastFrameTime < minFrameMs){
    requestAnimationFrame(gameLoop);
    return;
  }

  const dt = Math.min(.05,(now-lastFrameTime)/1000);
  lastFrameTime = now;

  if(gameStarted){
    const dtGame = dt*gameSpeed;
    gameElapsed += dtGame;
    fleetMenuUiAccumulator += dt;

    // Экономика, ИИ, исследования и захват не требуют 60 вычислений/сек.
    PERF.simulationAccumulator += dtGame;
    if(PERF.simulationAccumulator >= PERF.simulationStep){
      const simDt = Math.min(.75,PERF.simulationAccumulator);
      PERF.simulationAccumulator = 0;

      updateEconomy(simDt);
      updateEnemyAI(simDt);
      updateResearch(simDt);
      updateCaptures(simDt);
    }

    if(
      fleetMenuUiAccumulator >= 1 &&
      currentFleetPlanet &&
      planetFleetOverlay.classList.contains('open')
    ){
      fleetMenuUiAccumulator = 0;
      // Не пересобираем меню прямо под пальцем: это могло сбрасывать прокрутку
      // и уничтожать кнопку между pointerdown и click на медленных телефонах.
      if(performance.now() - planetMenuInteractionAt > 900){
        const savedScroll = planetMenuContent.scrollTop;
        planetFleetResources.innerHTML = planetHeaderSummaryHtml(currentFleetPlanet,true);
        renderPlanetMenuTab(currentFleetPlanet);
        planetMenuContent.scrollTop = savedScroll;
      }
    }

    if(resourceBarPlanet){
      updateSelectedPlanetResourceBar(false);
    }
  }

  // Движение кораблей остаётся плавным.
  updateScouts(dt);

  // Все дополнительные игровые системы работают через один центральный цикл.
  // Это заменяет независимые requestAnimationFrame-циклы модулей и уменьшает
  // лишнюю работу на мобильных устройствах.
  if(typeof window.AECentralTick === 'function'){
    try{ window.AECentralTick(gameStarted ? dt*gameSpeed : 0, now); }
    catch(err){ console.error('Central systems tick error:',err); }
  }

  // Тяжёлые DOM-обновления и проверка видимости выполняются реже.
  updateMiniDynamic(false);
  updateWorldCulling(false);

  requestAnimationFrame(gameLoop);
}

function prepareSignalLoot(a){
  if(a.loot) return a.loot;
  const loot={resources:{},ksenit:0,techId:null,commanderId:null,population:0,xp:22};
  switch(a.type){
    case 'derelict':
      loot.resources={'Железо':160+randi(0,70),'Медь':90+randi(0,45),'Литий':35+randi(0,25)};
      if(Math.random()<.42) loot.techId=remainingRareTechId();
      break;
    case 'ruins':
      loot.techId=remainingRareTechId(); loot.ksenit=35+randi(0,25); loot.xp=38;
      break;
    case 'rift':
      loot.techId=remainingRareTechId(); loot.ksenit=25+randi(0,25); loot.xp=45;
      break;
    case 'cache':
      loot.resources={'Никель':45+randi(0,25),'Титан':24+randi(0,15),'Серебро':20+randi(0,12),'Платина':10+randi(0,8)};
      break;
    case 'life':
      loot.ksenit=35+randi(0,25); loot.population=100+randi(0,90); loot.xp=35;
      break;
    case 'wreckfield':
      loot.resources={'Железо':100+randi(0,80),'Никель':35+randi(0,22),'Титан':18+randi(0,14),'Серебро':14+randi(0,10)};
      if(Math.random()<.30) loot.techId=remainingRareTechId();
      break;
    case 'beacon':
      loot.ksenit=45+randi(0,30); loot.commanderId=remainingCommanderId(); loot.xp=30;
      break;
    case 'prototype':
      loot.techId=remainingRareTechId(); loot.ksenit=20+randi(0,20); loot.xp=40;
      break;
    default:
      loot.resources={'Железо':120,'Медь':70};
  }
  if(!loot.techId && ['ruins','rift','prototype'].includes(a.type)) loot.ksenit+=55;
  if(!loot.commanderId && a.type==='beacon') loot.ksenit+=35;
  a.loot=loot;
  return loot;
}
function nearestPlayerColonyForSignal(ship){

  for(const p of planets){
    if(p.owner!=='player') continue;
    const d=Math.hypot(p.x-ship.x,p.y-ship.y);
    if(d<bestD){bestD=d;best=p;}
  }
  return best || playerPlanet;
}

  }

  let candidates = [];

  else if(selectedFleet.length) candidates = selectedFleet.filter(s=>s && s.owner==='player' && !s.destroyed);

  if(!candidates.length){
    if(a.targetedBy){
      const assigned=scouts.find(s=>s.id===a.targetedBy&&!s.destroyed);
      if(assigned) showHint('К сигналу уже летит «'+assigned.name+'». Дождитесь прибытия.');

    return;
  }
  if(candidates.length > 1){ showHint('Для исследования неизвестного сигнала выберите один корабль.'); return; }

  const s=candidates[0];
  if(s.cooldownState || s.scanState){ showHint('Этот корабль пока не готов к новой экспедиции.'); return; }
  if(s.resourceTransport){ showHint('Сначала завершите или отмените перевозку ресурсов.'); return; }

    if(old && old.targetedBy===s.id){ old.targetedBy=null; old.readyShipId=null; old.el?.classList.remove('targeted','ready'); }
  }

updateCaptures=function(dtGame){
  for(const planet of planets){

    const hostileOwner=ownerEnemy(planet.owner);

    defenseFire(planet,dtGame,invaders);
    siegeFireOnDefense(planet,dtGame,invaders,defenders);

    let c=planet.captureState;
    if(!c){
      if(invaders.length&&defenders.length===0&&!planetDefenseAlive(planet)) beginCapture(planet,hostileOwner);
      continue;
    }
    const capInvaders=siegeShipsForPlanet(planet,c.attackerOwner).filter(s=>Math.hypot(s.x-planet.x,s.y-planet.y)<=CAPTURE_RADIUS);
    const capDefenders=militaryNearPlanet(planet,planet.owner,CAPTURE_RADIUS+18);
    let reason='';
    if(planetDefenseAlive(planet)) reason='планетарная оборона восстановлена';
    else if(capDefenders.length) reason='на орбите снова есть флот защитника';
    else if(!capInvaders.length) reason='флот захвата покинул орбиту';
    if(reason){
      c.paused=true;c.pausedReason=reason;
      if(!capInvaders.length){c.absent=(c.absent||0)+dtGame;if(c.absent>=10){cancelCapture(planet,'атакующие корабли уничтожены или отступили');continue;}}
      else c.absent=0;
    }else{
      c.paused=false;c.pausedReason='';c.absent=0;c.remaining=Math.max(0,c.remaining-dtGame);
      const progress=1-c.remaining/c.total;planet.population=Math.max(0,Math.floor(c.startPopulation*(1-.28*progress)));
      for(const threshold of [120,60,30]) if(c.remaining<=threshold&&c.lastNotice>threshold){c.lastNotice=threshold;pushEvent('Захват «'+planet.name+'»: осталось '+threshold+' сек.',planet.owner==='player'?'attack':'discovery');break;}
    }
    updateCaptureBadge(planet);
    if(c.remaining<=0) finishCapture(planet,c);
  }
};
