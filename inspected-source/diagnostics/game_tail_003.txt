function reconcileEnemySpecialists(){
  for(const sp of PO.enemySpecialists){
    if(sp.owner!=='enemy') continue;
    if(sp.status==='stationed'){
      const station=planets.find(p=>p.id===sp.planetId&&p.owner==='enemy');
      if(!station){
        const fallback=nearestEnemyColonyForPersonnel(enemyPlanet?.x||0,enemyPlanet?.y||0);
        if(fallback) sp.planetId=fallback.id;
        else {sp.status='lost';sp.owner='none';sp.planetId=null;sp.shipId=null;}
      }
    }else if(sp.status==='transit'){
      const carrier=scouts.find(s=>s.id===sp.shipId&&!s.destroyed&&s._aePersonnelTransport?.specialistId===sp.id);

  }

      destination=nearestEnemyColonyForPersonnel(ship.x,ship.y);

        meta.destinationId=destination.id;

(()=>{

function selectedRangeShip(){
  if(typeof controlledShip!=='undefined' && controlledShip && !controlledShip.destroyed && controlledShip.owner==='player' && isMilitaryShipType(controlledShip.type)){
    return controlledShip;
  }
  if(typeof selectedFleet!=='undefined' && Array.isArray(selectedFleet) && selectedFleet.length===1){
    const s=selectedFleet[0];
    if(s && !s.destroyed && s.owner==='player' && isMilitaryShipType(s.type)) return s;
  }
  return null;

function hideAttackRangeCircle(){
  ring.style.display='none';
  ring.dataset.shipId='';
}

function updateAttackRangeCircle(ship=null){
  const target=ship && !ship.destroyed && ship.owner==='player' && isMilitaryShipType(ship.type)
    ? ship
    : selectedRangeShip();
  if(!target){ hideAttackRangeCircle(); return; }

  const range=Math.max(0,Number(target.attackRange)||0);
  if(range<=0){ hideAttackRangeCircle(); return; }

  const diameter=range*2;
  ring.style.display='block';
  ring.style.left=target.x+'px';
  ring.style.top=target.y+'px';
  ring.style.width=diameter+'px';
  ring.style.height=diameter+'px';
  ring.dataset.shipId=String(target.id||'');
  const rounded=Math.round(range*10)/10;
  label.textContent='Дальность '+(Number.isInteger(rounded)?rounded.toFixed(0):rounded.toFixed(1));
}
